    68→    def dispatch_block(self, block, input_ids, input_others):
    69→        """Multi-GPU aware block dispatch for SignRound tuning.
    70→
    71→        Stores _card_0_in_high_risk and _loss_device on self for use in quantize_block.
    72→        Returns the block after device placement.
    73→        """
    74→        from auto_round.utils import is_auto_device_mapping
    75→
    76→        if (
    77→            is_auto_device_mapping(device_manager.device_map)
    78→            and len(device_manager.device_list) > 1
    79→            and not self.model_context.is_diffusion
    80→        ):
    81→            from auto_round.utils.device import set_auto_device_map_for_block_with_tuning
    82→
    83→            card_0_in_high_risk, loss_device = set_auto_device_map_for_block_with_tuning(
    84→                block,
    85→                device_manager.device_list,
    86→                input_ids,
    87→                self.compress_context.low_gpu_mem_usage,
    88→                self.calibration_context.batch_size,
    89→                device_manager.device,
    90→            )
    91→            if len(device_manager.device_list) > 1:
    92→                from accelerate.hooks import AlignDevicesHook, add_hook_to_module
    93→
    94→                for _n, _mod in block.named_modules():
    95→                    if len(list(_mod.children())) != 0 or not hasattr(_mod, "tuning_device"):
    96→                        continue
    97→                    add_hook_to_module(_mod, AlignDevicesHook(_mod.tuning_device, io_same_device=True), True)
    98→        else:
    99→            block = block.to(device_manager.device)
   100→            card_0_in_high_risk, loss_device = False, device_manager.device
   101→
   102→        self._card_0_in_high_risk = card_0_in_high_risk
   103→        self._loss_device = loss_device
   104→        return block

   106→    def _get_non_zero_cnt(self, tensor: list[torch.Tensor], indices: list[int]) -> int:
   107→        current_tensors = [tensor[i] for i in indices]
   108→        non_zero_cnt = 0
   109→        for t in current_tensors:
   110→            non_zero_cnt += torch.count_nonzero(t).item()
   111→        return non_zero_cnt

   113→    def _get_loss(
   114→        self,
   115→        pred_output: torch.Tensor,
   116→        ref_output: torch.Tensor,
   117→        indices: torch.Tensor,
   118→        loss_func: Callable,
   119→        device: Union[str, torch.device] = "cpu",
   120→        valid_token_mask: Optional[torch.Tensor] = None,
   121→        input_ids=None,
   122→    ):
   123→        autocast_ctx = (
   124→            nullcontext()
   125→            if self.model_context.amp
   126→            else autocast(device_type=str(device).split(":")[0], dtype=self.model_context.amp_dtype)
   127→        )
   128→        if valid_token_mask:
   129→            tmp_attention_mask = [valid_token_mask[i] for i in indices]
   130→            tmp_attention_mask = torch.cat(tmp_attention_mask, dim=0).to(device)
   131→            tmp_attention_mask.unsqueeze_(-1)
   132→
   133→            with autocast_ctx:
   134→                loss = loss_func(  # pylint: disable=not-callable
   135→                    (pred_output * tmp_attention_mask).to(torch.float32),
   136→                    (ref_output * tmp_attention_mask).to(torch.float32),
   137→                )
   138→        else:
   139→            with autocast_ctx:
   140→                loss = loss_func(  # pylint: disable=not-callable
   141→                    pred_output.to(torch.float32), ref_output.to(torch.float32)
   142→                )
   143→
   144→        return loss

   146→    def _count_samples(self, inputs: Any) -> int:
   147→        if isinstance(inputs, dict):
   148→            hs = inputs.get("hidden_states")
   149→            return len(hs) if isinstance(hs, list) else hs.shape[self.calibration_context.batch_dim]
   150→        elif isinstance(inputs, list):
   151→            return len(inputs)
   152→        else:
   153→            return inputs.shape[self.calibration_context.batch_dim]

   155→    def _init_lm_components(self) -> None:
   156→        """Lazily locate and cache ``_post_block_modules`` and ``_lm_head`` for LFQ loss.
   157→
   158→        ``_post_block_modules`` is an ordered list of modules that must be applied
   159→        between the last transformer-block output and the lm_head projection.
   160→        Most architectures have a single final norm; OPT additionally has an
   161→        optional ``project_out`` projection.
   162→
   163→        Supported without manual configuration:
   164→            LLaMA / Qwen / Gemma / Mistral / InternLM / Phi-3 — ``model.model.norm``
   165→            OPT — ``model.model.decoder.{final_layer_norm, project_out}`` (both optional)
   166→            GPT-2 / Falcon / Bloom — ``model.transformer.ln_f``
   167→            GPT-NeoX / Pythia — ``model.gpt_neox.final_layer_norm``
   168→            Phi / Phi-2 — ``model.model.final_layernorm``
   169→            MPT — ``model.transformer.norm_f``
   170→            ChatGLM — ``model.transformer.encoder.final_layernorm``
   171→            RWKV — ``model.rwkv.ln_out``
   172→
   173→        Raises ``AttributeError`` if no lm_head equivalent can be found.
   174→        """
   175→        if hasattr(self, "_lm_head"):
   176→            return
   177→
   178→        model = self.model
   179→
   180→        # ── lm_head ──────────────────────────────────────────────────────────
   181→        for name in ("lm_head", "embed_out", "output", "head"):
   182→            if hasattr(model, name):
   183→                self._lm_head = getattr(model, name)
   184→                break
   185→        else:
   186→            raise AttributeError(
   187→                f"Cannot locate lm_head in {type(model).__name__}. " "Checked: lm_head, embed_out, output, head."
   188→            )
   189→
   190→        # ── post-block processing (ordered list applied before lm_head) ───────
   191→        # OPT: decoder has both an optional final_layer_norm *and* an optional
   192→        # project_out that maps ffn_dim → word_embed_proj_dim.  Detect by probing
   193→        # for the characteristic project_out attribute (may be None).
   194→        try:
   195→            decoder = model.model.decoder
   196→            _ = decoder.project_out  # raises AttributeError if not an OPT decoder
   197→            self._post_block_modules = [m for m in (decoder.final_layer_norm, decoder.project_out) if m is not None]
   198→            return
   199→        except AttributeError:
   200→            pass
   201→
   202→        # All other architectures: single optional final norm.
   203→        norm_getters = [
   204→            lambda: model.model.norm,  # LLaMA / Qwen / Gemma / Mistral / InternLM / Phi-3
   205→            lambda: model.transformer.ln_f,  # GPT-2 / Falcon / Bloom
   206→            lambda: model.gpt_neox.final_layer_norm,  # GPT-NeoX / Pythia
   207→            lambda: model.model.final_layernorm,  # Phi / Phi-2
   208→            lambda: model.transformer.norm_f,  # MPT
   209→            lambda: model.transformer.encoder.final_layernorm,  # ChatGLM
   210→            lambda: model.rwkv.ln_out,  # RWKV
   211→        ]
   212→        self._post_block_modules = []
   213→        for getter in norm_getters:
   214→            try:
   215→                norm = getter()
   216→                if norm is not None:
   217→                    self._post_block_modules = [norm]
   218→                    break
   219→            except AttributeError:
   220→                continue

   244→    def _is_text_decoder_block(self, block_name: str) -> bool:
   245→        """Return ``True`` if *block_name* refers to a text-decoder block.
   246→
   247→        Blocks whose names contain any of the non-text keywords (vision, audio,
   248→        image, …) are considered multimodal and excluded from LFQ loss.
   249→        """
   250→        name_lower = block_name.lower()
   251→        return not any(kw in name_lower for kw in self._NON_TEXT_BLOCK_KEYWORDS)

   253→    def lfq_loss(self, hidden_state: torch.Tensor, input_ids: torch.Tensor) -> torch.Tensor:
   254→        """Compute LM cross-entropy loss from the last block's hidden states.
   255→
   256→        Applies every post-block module (final norm, optional projection, …) in
   257→        order, then runs lm_head and computes next-token prediction loss.
   258→        Positions marked with ``-100`` in *input_ids* are excluded from the loss.
   259→
   260→        Args:
   261→            hidden_state: Last block output, shape ``[batch, seq_len, hidden]``.
   262→            input_ids:    Token-ID labels with ``-100`` for ignored positions,
   263→                          shape ``[batch, seq_len]``.
   264→
   265→        Returns:
   266→            Scalar cross-entropy loss tensor.
   267→        """
   268→        self._init_lm_components()
   269→        device = hidden_state.device
   270→
   271→        for module in self._post_block_modules:
   272→            module.to(device)
   273→            hidden_state = module(hidden_state)
   274→
   275→        self._lm_head.to(device)
   276→        logits = self._lm_head(hidden_state)
   277→
   278→        if hasattr(self.model, "loss_function"):
   279→            loss = self.model.loss_function(
   280→                logits=logits,
   281→                labels=input_ids.to(device),
   282→                vocab_size=self.model.config.vocab_size,
   283→            )
   284→        else:
   285→            import torch.nn.functional as F
   286→
   287→            # Standard causal-LM shift: predict token t+1 from hidden state t.
   288→            shift_logits = logits[..., :-1, :].contiguous()
   289→            shift_labels = input_ids[..., 1:].contiguous().to(device)
   290→            loss = F.cross_entropy(
   291→                shift_logits.view(-1, shift_logits.size(-1)),
   292→                shift_labels.view(-1),
   293→                ignore_index=-100,
   294→            )
   295→        return loss

   297→    def quantize_block(
   298→        self,
   299→        block,
   300→        fp_inputs,
   301→        input_others,
   302→        fp_outputs,
   303→        q_inputs,
   304→        block_ctx,
   305→        input_ids=None,
   306→        **kwargs,
   307→    ) -> dict:
   308→        """Apply the AutoRound optimization algorithm to a block.
   309→
   310→        This is the pure-algorithm entry point.  All infrastructure concerns
   311→        (device placement, act-max hook collection, DDP setup, memory cleanup,
   312→        logging) are handled by the Compressor before and after this call.
   313→
   314→        Args:
   315→            block: The transformer block module to quantize.
   316→            fp_inputs: FP calibration inputs for this block (list[Tensor] or dict
   317→                for diffusion models).
   318→            input_others: Auxiliary kwargs passed to the block forward
   319→                (e.g. attention_mask, position_ids).
   320→            fp_outputs: FP reference outputs of the block used as the optimization
   321→                target for the sign-gradient descent loss (list[Tensor]).
   322→            q_inputs: Quantized inputs from the previous block, or ``None`` when
   323→                cascaded quantized-input is disabled.
   324→            block_ctx: Per-block pipeline context (BlockContext).
   325→            input_ids: Raw token IDs from the tokenizer (``[1, seq_len]`` per
   326→                sample). Used to derive the valid-token loss mask once (result
   327→                cached on ``self._cached_valid_token_mask`` for reuse across
   328→                all blocks). ``None`` disables loss masking.
   329→            **kwargs: Reserved for forward-compatibility with future parameters.
   330→
   331→        Returns:
   332→            dict: Best quantization parameters found during optimization, or an
   333→                empty dict if no trainable parameters were found.
   334→        """
   335→        device = device_manager.device
   336→        loss_device = getattr(self, "_loss_device", device)
   337→        card_0_in_high_risk = getattr(self, "_card_0_in_high_risk", False)
   338→        mid_iter_mem_check = self.compress_context.low_gpu_mem_usage and card_0_in_high_risk
   339→
   340→        valid_token_mask = None
   341→        # Derive valid_token_mask from raw token IDs when not supplied by caller.
   342→        # Result is cached on self so it is computed only once across all blocks.
   343→        if input_ids is not None:
   344→            if not hasattr(self, "_cached_valid_token_mask"):
   345→                self._cached_valid_token_mask = self._compute_valid_token_mask(input_ids)
   346→            valid_token_mask = self._cached_valid_token_mask
   347→
   348→        # Use quantized inputs if available and enabled
   349→        active_inputs = q_inputs if (q_inputs is not None and self.enable_quanted_input) else fp_inputs
   350→        nsamples = len(active_inputs) if isinstance(active_inputs, list) else self._count_samples(active_inputs)
   351→
   352→        quantized_layer_names, unquantized_layer_names = self.wrapper_block(
   353→            block,
   354→            self.enable_minmax_tuning,
   355→            self.enable_norm_bias_tuning,
   356→            enable_torch_compile=self.compress_context.enable_torch_compile,
   357→            device=device,
   358→        )
   359→
   360→        round_params = []
   361→        minmax_params = []
   362→        for n, m in block.named_modules():
   363→            if hasattr(m, "orig_layer"):
   364→                for key in m.params.keys():
   365→                    if "min" in key or "max" in key:
   366→                        minmax_params.append(m.params[key])
   367→                    else:
   368→                        round_params.append(m.params[key])
   369→
   370→        lr = torch.tensor(self.lr)
   371→        minmax_lr = torch.tensor(self.minmax_lr)
   372→
   373→        extra_kwargs = {} if self.momentum is None else {"momentum": self.momentum}
   374→
   375→        if self.enable_minmax_tuning:
   376→            params = [
   377→                {"params": round_params},
   378→                {"params": minmax_params, "lr": minmax_lr},
   379→            ]
   380→        else:
   381→            params = round_params
   382→
   383→        optimizer = self.optimizer(
   384→            params,
   385→            lr=lr,
   386→            weight_decay=0,
   387→            **extra_kwargs,
   388→        )
   389→
   390→        if len(round_params) + len(minmax_params) <= 0:
   391→            dump_info = (
   392→                f"quantized {len(quantized_layer_names)}/{(len(quantized_layer_names) + len(unquantized_layer_names))} "
   393→                f"layers in the block"
   394→            )
   395→            logger.info(dump_info)
   396→            unwrapper_block(block, {})
   397→            return {}
   398→
   399→        if self.lr_scheduler is None:
   400→            lr_schedule = torch.optim.lr_scheduler.LinearLR(
   401→                optimizer, start_factor=1.0, end_factor=0.0, total_iters=self.iters
   402→            )
   403→        else:
   404→            lr_schedule = copy.deepcopy(self.lr_scheduler)
   405→
   406→        last_best_iter = 0
   407→        best_loss = torch.finfo(torch.float).max
   408→        num_elm = 1
   409→        mse_reduction = "mean"
   410→        if self.gradient_accumulate_steps != 1:
   411→            mse_reduction = "sum"
   412→        mse_loss = torch.nn.MSELoss(reduction=mse_reduction).to(device)
   413→        scaler = self._get_scaler()  # pylint: disable=assignment-from-none
   414→        init_loss = None
   415→        best_params = {}
   416→        total_loss = 0
   417→        batch_size = self.calibration_context.batch_size
   418→        global_batch_size = batch_size * self.gradient_accumulate_steps
   419→        global_batch_size = min(nsamples, global_batch_size)
   420→        # Compute num_elm once before the loop (used to normalise the accumulated loss).
   421→        # We assume the block input and output shape is same
   422→        if self.gradient_accumulate_steps != 1 and not valid_token_mask:
   423→            whole_indices = torch.arange(global_batch_size)
   424→            if isinstance(active_inputs, list):  # dict for diffusion, tricky setting, not sure whether it's correct
   425→                num_elm = sum(active_inputs[i.item()].numel() for i in whole_indices)
   426→
   427→        block, sync_gradients = setup_ddp_if_needed_(self, block, device_manager.device_list)
   428→        index_sampler = IndexSampler(nsamples, global_batch_size)
   429→        block_fwd = self.block_forward
   430→
   431→        # When low_gpu_mem_usage is enabled, active_inputs / fp_outputs are intentionally
   432→        # kept on CPU to limit GPU memory.  However block_fwd normally routes pred_output
   433→        # through CPU (cache_device="cpu") and the very next line moves it back to
   434→        # loss_device — a wasteful GPU→CPU→GPU roundtrip on every batch × iteration.
   435→        # pred_output is a transient single-batch tensor consumed immediately for the
   436→        # loss and then freed, so keeping it on the compute device costs no persistent
   437→        # extra memory.  Pass it as a per-call override so self.cache_device is unchanged.
   438→        _fwd_cache_device = (
   439→            device
   440→            if getattr(self.compress_context, "low_gpu_mem_usage", False) and not str(device).startswith("cpu")
   441→            else None
   442→        )
   443→
   444→        for i in range(self.iters):
   445→            if self.enable_alg_ext and self.scheme.data_type.endswith("dq"):
   446→                for n, m in block.named_modules():
   447→                    m.cur_iter = i
   448→            total_loss = 0
   449→            global_indices = index_sampler.next_batch()
   450→            if valid_token_mask:
   451→                num_elm = self._get_non_zero_cnt(valid_token_mask, global_indices)
   452→
   453→            for batch_start in range(0, len(global_indices), batch_size):
   454→                indices = global_indices[batch_start : batch_start + batch_size]
   455→                ref_output = torch.cat([fp_outputs[i] for i in indices], dim=0).to(loss_device)
   456→                pred_output = block_fwd.forward(block, active_inputs, input_others, indices, _fwd_cache_device)
   457→                if loss_device is not None:
   458→                    pred_output = pred_output.to(loss_device)
   459→                if (
   460→                    block_ctx.block_index == block_ctx.block_cnt - 1
   461→                    and self.enable_lfq
   462→                    and input_ids is not None
   463→                    and self._is_text_decoder_block(block_ctx.block_name)
   464→                ):
   465→                    loss = self.lfq_loss(pred_output, torch.cat([input_ids[i] for i in indices], dim=0))
   466→                else:
   467→                    loss = self._get_loss(pred_output, ref_output, indices, mse_loss, device, valid_token_mask)
   468→                num_elm = 1 if num_elm <= 0 else num_elm
   469→                total_loss += loss.item() / num_elm
   470→
   471→                if mid_iter_mem_check:
   472→                    # clear memory to avoid OOM due to memory fragmentation
   473→                    clear_memory_if_reached_threshold(threshold=0.5, device_list=device_manager.device_list)
   474→
   475→                self._scale_loss_and_backward(scaler, loss)
   476→
   477→                if mid_iter_mem_check:
   478→                    # clear memory to avoid OOM due to memory fragmentation
   479→                    clear_memory_if_reached_threshold(threshold=0.8, device_list=device_manager.device_list)
   480→
   481→            if i == 0:
   482→                init_loss = total_loss
   483→
   484→            if total_loss < best_loss:
   485→                best_loss = total_loss
   486→                if not self.not_use_best_mse:
   487→                    best_params = collect_best_params(block, self.compress_context.cache_device)
   488→                    last_best_iter = i
   489→            if self.not_use_best_mse and i == self.iters - 1:
   490→                best_params = collect_best_params(block, self.compress_context.cache_device)
   491→
   492→            if not self.not_use_best_mse:
   493→                if 0 < self.dynamic_max_gap <= i - last_best_iter:
   494→                    break
   495→            sync_gradients()
   496→            self._step(scaler, optimizer, lr_schedule)
   497→
   498→        last_loss = total_loss
   499→        best_iter = self.iters
   500→        if not self.not_use_best_mse:
   501→            last_loss = best_loss
   502→            best_iter = last_best_iter
   503→        if self.iters > 0:
   504→            dump_info = (
   505→                f"quantized {len(quantized_layer_names)}/{(len(quantized_layer_names) + len(unquantized_layer_names))} "
   506→                f"layers in the block, loss iter 0: {init_loss:.6f} -> iter {best_iter}: {last_loss:.6f}"
   507→            )
   508→        else:
   509→            dump_info = (
   510→                f"quantized {len(quantized_layer_names)}/{(len(quantized_layer_names) + len(unquantized_layer_names))} "
   511→                "layers in the block"
   512→            )
   513→
   514→        self.compress_context.clear_memory()  # clear cached memory during training
   515→        if len(unquantized_layer_names) != 0:
   516→            logger.info(f"Unquantized layers: {unquantized_layer_names}")
   517→        with torch.no_grad():
   518→            unwrapper_block(block, best_params)
   519→
   520→        if self.config.is_act_nv_fp:
   521→            # enable moe experts act_max automatic generation for WrapperWALayer
   522→            set_amax_for_all_moe_layers(block, attr_name="orig_layer.act_max")
   523→
   524→        logger.infoclean(dump_info)
   525→        return best_params
   526→
   527→    def quantize_layer_outside_block(
   528→        self,
   529→        layer: "torch.nn.Module",
   530→        fp_inputs: Optional[list[torch.Tensor]] = None,
   531→        q_inputs: Optional[list[torch.Tensor]] = None,
   532→        disable_opt_rtn: Optional[bool] = None,
   533→        input_ids: Optional[list[torch.Tensor]] = None,
   534→    ):
   535→        """Quantize a single layer that lives outside a transformer block.
   536→
   537→        When ``fp_inputs`` is provided the layer is tuned with the sign-gradient
   538→        descent optimizer (same loss loop as block-level quantization).  When
   539→        ``fp_inputs`` is ``None`` the method falls back to zero-shot RTN.
   540→
   541→        Args:
   542→            layer: The layer module to quantize.  Must have a ``global_name``
   543→                attribute for model re-insertion and logging.
   544→            fp_inputs: Per-sample FP activations fed into this layer, used as
   545→                calibration inputs during optimization. ``None`` triggers RTN
   546→                fallback.
   547→            q_inputs: Per-sample quantized activations from the previous stage,
   548→                used instead of ``fp_inputs`` during the forward pass when
   549→                cascaded quantized-input is enabled. ``None`` means use
   550→                ``fp_inputs`` for both reference and tuning forward.
   551→            disable_opt_rtn: Override optimized-RTN; ``None`` defers to quantizer config.
   552→            input_ids: Raw token IDs from the tokenizer (``[1, seq_len]`` per
   553→                sample); used to derive the valid-token loss mask via
   554→                ``_compute_valid_token_mask``. ``None`` disables loss masking.
   555→        """
   556→
   557→        layer_name = layer.global_name
   558→        if fp_inputs is None:
   559→            logger.info(f"using rtn to quantize {layer_name}")
   560→            self._quantize_layer_via_rtn(
   561→                layer,
   562→                disable_opt_rtn=(
   563→                    disable_opt_rtn if disable_opt_rtn is not None else getattr(self.config, "disable_opt_rtn", True)
   564→                ),
   565→            )
   566→            return
   567→
   568→        # Derive valid_token_mask from raw token IDs when not supplied by caller.
   569→        # Reuse the cached mask if already computed by a previous block.
   570→        valid_token_mask = None
   571→        if input_ids is not None:
   572→            if not hasattr(self, "_cached_valid_token_mask"):
   573→                self._cached_valid_token_mask = self._compute_valid_token_mask(input_ids)
   574→            valid_token_mask = self._cached_valid_token_mask
   575→
   576→        logger.info(f"quantizing layer {layer_name}")
   577→        # Layer is already on the correct device (placed by the caller / AlgorithmComposer).
   578→        device = layer.weight.device if hasattr(layer, "weight") else device_manager.device
   579→        for i in range(len(fp_inputs)):
   580→            fp_inputs[i] = fp_inputs[i].to(layer.weight.dtype)
   581→            if q_inputs is not None:
   582→                q_inputs[i] = q_inputs[i].to(layer.weight.dtype)
   583→
   584→        wrapper_linear = WrapperLinear(
   585→            layer,
   586→            enable_minmax_tuning=self.enable_minmax_tuning,
   587→            enable_torch_compile=self.compress_context.enable_torch_compile,
   588→            device=device,
   589→        ).to(device)
   590→        round_params = []
   591→        minmax_params = []
   592→        for key in wrapper_linear.params.keys():
   593→            if "min" in key or "max" in key:
   594→                minmax_params.append(wrapper_linear.params[key])
   595→            else:
   596→                round_params.append(wrapper_linear.value)
   597→        if len(round_params) + len(minmax_params) <= 0:
   598→            dump_info = f"quantized {layer_name}"
   599→            logger.info(dump_info)
   600→            with torch.no_grad():
   601→                unwrapper_layer(self.model, wrapper_linear, layer_name, {})
   602→            mv_module_from_gpu(layer)
   603→
   604→        lr = torch.tensor(self.lr)
   605→        minmax_lr = torch.tensor(self.minmax_lr)
   606→        if self.enable_minmax_tuning:
   607→            optimizer = self.optimizer(
   608→                [{"params": round_params}, {"params": minmax_params, "lr": minmax_lr}], lr=lr, weight_decay=0
   609→            )
   610→        else:
   611→            optimizer = self.optimizer(round_params, lr=lr, weight_decay=0)
   612→
   613→        if self.lr_scheduler is None:
   614→            lr_schedule = torch.optim.lr_scheduler.LinearLR(
   615→                optimizer, start_factor=1.0, end_factor=0.0, total_iters=self.iters
   616→            )
   617→        else:
   618→            lr_schedule = copy.deepcopy(self.lr_scheduler)
   619→        nsamples = len(fp_inputs)
   620→        last_best_iter = 0
   621→        best_loss = torch.finfo(torch.float).max
   622→        best_params = None
   623→        scaler = self._get_scaler()  # pylint: disable=assignment-from-none
   624→        init_loss = None
   625→
   626→        gradient_accumulate_steps = (
   627→            self.calibration_context.batch_size * self.gradient_accumulate_steps
   628→        )  # Force to low gpu
   629→
   630→        total_loss = 0
   631→        num_elm = 1
   632→        mse_reduction = "mean"
   633→        if gradient_accumulate_steps != 1:
   634→            mse_reduction = "sum"
   635→        mse_loss = torch.nn.MSELoss(reduction=mse_reduction).to(device)
   636→        batch_size = 1  # Force to low gpu
   637→        global_batch_size = gradient_accumulate_steps
   638→        global_batch_size = min(nsamples, global_batch_size)
   639→        # Compute num_elm once before the loop.
   640→        if gradient_accumulate_steps != 1:
   641→            whole_indices = list(range(global_batch_size))
   642→            if valid_token_mask:
   643→                num_elm = self._get_non_zero_cnt(valid_token_mask, whole_indices)
   644→            elif q_inputs is not None:
   645→                num_elm = self._count_layer_input_elements(q_inputs, whole_indices)
   646→            else:
   647→                num_elm = self._count_layer_input_elements(fp_inputs, whole_indices)
   648→
   649→        index_sampler = IndexSampler(nsamples, global_batch_size)
   650→
   651→        for i in range(self.iters):
   652→            total_loss = 0
   653→            global_indices = index_sampler.next_batch()
   654→
   655→            for batch_start in range(0, len(global_indices), batch_size):
   656→                indices = global_indices[batch_start : batch_start + batch_size]
   657→                if q_inputs is not None:
   658→                    current_input = [q_inputs[i] for i in indices]
   659→                    current_input = torch.cat(current_input, dim=0).to(device)
   660→                    org_input = [fp_inputs[i] for i in indices]
   661→                    org_input = torch.cat(org_input, dim=0).to(device)
   662→                else:
   663→                    current_input = [fp_inputs[i] for i in indices]
   664→                    current_input = torch.cat(current_input, dim=0).to(device)
   665→                    org_input = current_input
   666→                with torch.no_grad():
   667→                    current_output = layer(org_input)
   668→                autocast_ctx = (
   669→                    nullcontext()
   670→                    if not self.model_context.amp
   671→                    else autocast(device_type=str(device).split(":")[0], dtype=self.model_context.amp_dtype)
   672→                )
   673→                if valid_token_mask:
   674→                    tmp_valid_mask = [valid_token_mask[i] for i in indices]
   675→                    tmp_valid_mask = torch.cat(tmp_valid_mask, dim=0).to(device)
   676→                    tmp_valid_mask.unsqueeze_(-1)
   677→
   678→                    with autocast_ctx:
   679→                        output_q = wrapper_linear(current_input)  # pylint: disable=not-callable
   680→                        loss = mse_loss(  # pylint: disable=not-callable
   681→                            (output_q * tmp_valid_mask).to(torch.float32),
   682→                            (current_output * tmp_valid_mask).to(torch.float32),
   683→                        )
   684→
   685→                else:
   686→                    with autocast_ctx:
   687→                        output_q = wrapper_linear(current_input)  # pylint: disable=not-callable
   688→                        loss = mse_loss(  # pylint: disable=not-callable
   689→                            output_q.to(torch.float32),
   690→                            current_output.to(torch.float32),  # mul 1.0 will copy the output
   691→                        )
   692→
   693→                num_elm = 1 if num_elm <= 0 else num_elm
   694→                total_loss += loss.item() / num_elm
   695→
   696→                self._scale_loss_and_backward(scaler, loss)
   697→            if i == 0:
   698→                init_loss = total_loss
   699→
   700→            if total_loss < best_loss:
   701→                best_loss = total_loss
   702→                if not self.not_use_best_mse:
   703→                    best_params = collect_best_params(wrapper_linear, self.compress_context.cache_device)
   704→                    last_best_iter = i
   705→            if self.not_use_best_mse and i == self.iters - 1:
   706→                best_params = collect_best_params(wrapper_linear, self.compress_context.cache_device)
   707→
   708→            if not self.not_use_best_mse:
   709→                if 0 < self.dynamic_max_gap <= i - last_best_iter:
   710→                    break
   711→            self._step(scaler, optimizer, lr_schedule)
   712→
   713→        last_loss = total_loss
   714→        best_iter = self.iters
   715→        if not self.not_use_best_mse:
   716→            last_loss = best_loss
   717→            best_iter = last_best_iter
   718→        with torch.no_grad():
   719→            unwrapper_layer(self.model, wrapper_linear, layer_name, best_params)
   720→        mv_module_from_gpu(layer)
   721→        dump_info = f"quantized {layer_name},  loss iter 0: {init_loss:.6f} -> iter {best_iter}: {last_loss:.6f}"
   722→        logger.info(dump_info)
   723→
   724→    def finalize_run(self) -> None:
   725→        """Clear per-run caches (``_cached_valid_token_mask``, LFQ components)."""
   726→        for attr in ("_cached_valid_token_mask", "_lm_head", "_post_block_modules"):
   727→            if hasattr(self, attr):
   728→                delattr(self, attr)

   730→    def _get_optimizer(self, optimizer: Any):
   731→        """Returns the specified optimizer. In SignRound, we fix the optimizer.
   732→
   733→        Args:
   734→        optimizer: The optimizer to be used.
   735→
   736→        Returns:
   737→        The specified optimizer.
   738→        """
   739→        if optimizer is not None:
   740→            logger.warning_once(
   741→                "The optimizer setting in config will be ignored in AutoRound, using SignSGD as default."
   742→            )
   743→        return SignSGD

   745→    def _count_layer_input_elements(self, input_ids, indices: list) -> int:
   746→        return sum(input_ids[i].numel() for i in indices)
   747→
   748→    def _get_scaler(self):
   749→        """Returns scaler, in SignRound, no need to use scaler."""
   750→        return None

   752→    def _scale_loss_and_backward(self, scaler: Any, loss: torch.Tensor) -> torch.Tensor:
   753→        """Scales the loss and performs backward pass.
   754→
   755→        Args:
   756→        scaler: The scaler to be used.
   757→        loss: The loss to be scaled.
   758→
   759→        Returns:
   760→        The scaled loss.
   761→        """
   762→        scale_loss = loss * 1000
   763→        scale_loss.backward()
   764→        if is_hpex_available():
   765→            htcore.mark_step()
   766→        return scale_loss
   767→
   768→    def _step(self, scaler: Any, optimizer: Any, lr_schedule: Any):
   769→        """Performs a step in the optimization process.
   770→
   771→        Args:
   772→        scaler: The scaler to be used.
   773→        optimizer: The optimizer for the step.
   774→        lr_schedule: The learning rate schedule.
   775→
   776→        Returns:
   777→        None
   778→        """
   779→        optimizer.step()
   780→        # for hpu
   781→        if is_hpex_available():
   782→            htcore.mark_step()
   783→        optimizer.zero_grad()
   784→        lr_schedule.step()
