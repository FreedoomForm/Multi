    39→def get_scale_shape(weight, group_size):
    40→    """Computes the shape of the scale tensor for quantization based on the weight tensor and group size.
    41→
    42→    Args:
    43→      weight (torch.Tensor): The weight tensor of the layer.
    44→      group_size (int or tuple): The size of the groups for quantization.
    45→
    46→    Returns:
    47→      The shape of the scale tensor to be used for quantization.
    48→    """
    49→    if isinstance(group_size, tuple):
    50→        assert len(weight.shape) == len(group_size), f"Expected group_size is {len(weight.shape)}D but get {group_size}"
    51→        return (weight.shape[0] // group_size[0], weight.shape[1] // group_size[1])
    52→    if group_size == 0:
    53→        return 1
    54→    elif group_size == -1 or weight.shape[1] < group_size:
    55→        shape = weight.shape[0]
    56→    else:
    57→        shape = weight.shape[0] * ceil(weight.shape[1] / group_size)
    58→
    59→    return shape

    62→class WrapperLinear(torch.nn.Module):
    63→    """A wrapper for linear/conv1d layers to enable quantization and tuning.
    64→
    65→    This module wraps an existing linear or conv1d layer and provides additional functionality
    66→    for quantization, parameter tuning, and activation/bias normalization.
    67→
    68→    Args:
    69→        orig_layer (torch.nn.Module): The original layer to be wrapped (linear or conv1d).
    70→        enable_minmax_tuning (bool): Whether to enable min-max scale tuning.
    71→        enable_norm_bias_tuning (bool): Whether to enable normalization and tuning of the bias term.
    72→        enable_torch_compile (bool): Whether to enable torch compilation.
    73→        device (str): Device on which to run computations (e.g., 'cpu' or 'cuda').
    74→    """
    75→
    76→    minmax_scale_bound = (0.0, 1.0)
    77→
    78→    def __init__(
    79→        self,
    80→        orig_layer,
    81→        enable_minmax_tuning=True,
    82→        enable_norm_bias_tuning=False,
    83→        device="cpu",
    84→        enable_round_tuning=True,
    85→        enable_torch_compile=True,
    86→        disable_opt_rtn=True,
    87→        **kwargs,
    88→    ):
    89→        """Initializes the WrapperLinear module.
    90→
    91→        Args:
    92→            orig_layer (torch.nn.Module): The original layer to wrap.
    93→            enable_minmax_tuning (bool): Whether to enable min-max scale tuning.
    94→            enable_norm_bias_tuning (bool): Whether to enable normalization and tuning for the bias term.
    95→            device (str): The computation device, such as 'cpu' or 'cuda'.
    96→        """
    97→        super(WrapperLinear, self).__init__()
    98→        self.orig_layer = orig_layer
    99→        self.orig_layer.iters = kwargs.pop("iters", 200)
   100→        self.disable_opt_rtn = disable_opt_rtn
   101→        self.output_device = device
   102→        self.device = self.orig_layer.tuning_device if hasattr(self.orig_layer, "tuning_device") else device
   103→        self.enable_minmax_tuning = enable_minmax_tuning
   104→        self.enable_round_tuning = enable_round_tuning
   105→        self.enable_torch_compile = enable_torch_compile
   106→        self.enable_norm_bias_tuning = enable_norm_bias_tuning and (orig_layer.bias is not None)
   107→        self.enable_act_quant = self.orig_layer.act_bits <= 8
   108→        self.weight_global_scale = getattr(self.orig_layer, "weight_global_scale", None)
   109→        if is_nv_fp(self.orig_layer.data_type) and self.weight_global_scale is None:
   110→            from auto_round.data_type.nvfp import calculate_gparam
   111→
   112→            weight_global_scale = calculate_gparam(self.orig_layer.weight, self.orig_layer.group_size)
   113→            setattr(self, "weight_global_scale", weight_global_scale)
   114→            self.weight_global_scale = self.weight_global_scale.to(self.orig_layer.weight.device)
   115→        if hasattr(self.orig_layer, "scale_dtype") and self.orig_layer.scale_dtype == torch.float32:
   116→            self.q_scale_thresh = 1e-8
   117→        else:
   118→            self.q_scale_thresh = 1e-5
   119→        self._init_tuning_params_and_quant_func()
   120→        if deepspeed_exists:
   121→            if type(self.orig_layer) in (torch.nn.Linear, LinearLayer):
   122→                self.orig_forward = self.linear_forward
   123→            elif type(self.orig_layer) == LinearAllreduce:
   124→                self.orig_forward = self.all_reduce_linear_forward
   125→                self.mp_group = self.orig_layer.mp_group
   126→            else:
   127→                self.orig_forward = self.conv1d_forward
   128→        else:
   129→            self.orig_forward = self.linear_forward if type(self.orig_layer) == torch.nn.Linear else self.conv1d_forward
   130→
   131→    @property
   132→    def weight(self):
   133→        return self.orig_layer.weight
   134→
   135→    @property
   136→    def bias(self):
   137→        return self.orig_layer.bias

   139→    def _init_tuning_params_and_quant_func(self):
   140→        """Initializes tuning parameters and quantization functions.
   141→
   142→        This method sets up required parameters and functions for weight quantization,
   143→        activation quantization, and bias/normalization.
   144→        """
   145→        self.params = {}
   146→        p_dtype = torch.float32  ##parameter dtype
   147→
   148→        orig_layer = self.orig_layer
   149→        orig_weight = getattr(orig_layer, "get_weight", lambda: orig_layer.weight)()
   150→        if type(self.orig_layer) == transformers.pytorch_utils.Conv1D:
   151→            orig_weight = orig_weight.t()
   152→        weight_reshape, _, _ = reshape_pad_tensor_by_group_size(orig_weight.data, orig_layer.group_size)
   153→
   154→        if self.enable_round_tuning:
   155→            self.weight_min = (
   156→                torch.clamp(weight_reshape.amin(dim=(-2, -1)), max=0)
   157→                if isinstance(orig_layer.group_size, tuple)
   158→                else torch.clamp(weight_reshape.min(1)[0], max=0)
   159→            )
   160→            self.weight_max = (
   161→                torch.clamp(weight_reshape.amax(dim=(-2, -1)), min=0)
   162→                if isinstance(orig_layer.group_size, tuple)
   163→                else torch.clamp(weight_reshape.max(1)[0], min=0)
   164→            )
   165→        else:
   166→            self.weight_min = None
   167→            self.weight_max = None
   168→        # AWQ clip-as-init: cap the tunable weight range to the per-group clip
   169→        # range searched by AWQ (``apply_clip`` with ``clip_as_init=True``).
   170→        # This initializes the range used by quant_tensor_sym/asym, leaving
   171→        # min_scale/max_scale to tune a coefficient on top. Only the standard
   172→        # (non-tuple) group layout maps onto weight_min/weight_max here.
   173→        awq_clip_min = getattr(orig_layer, "awq_clip_min", None)
   174→        awq_clip_max = getattr(orig_layer, "awq_clip_max", None)
   175→        if awq_clip_max is not None and self.weight_min is not None and not isinstance(orig_layer.group_size, tuple):
   176→            clip_max_flat = awq_clip_max.reshape(-1).to(self.weight_max.device, self.weight_max.dtype)
   177→            if awq_clip_min is None:
   178→                clip_min_flat = -clip_max_flat
   179→            else:
   180→                clip_min_flat = awq_clip_min.reshape(-1).to(self.weight_min.device, self.weight_min.dtype)
   181→            if clip_max_flat.numel() == self.weight_max.numel() and clip_min_flat.numel() == self.weight_min.numel():
   182→                self.weight_max = torch.minimum(self.weight_max, clip_max_flat)
   183→                self.weight_min = torch.maximum(self.weight_min, clip_min_flat)
   184→        self._init_params(
   185→            "value", p_dtype, weight_reshape.shape, 0, self.enable_round_tuning and self.orig_layer.bits < 16
   186→        )
   187→        # Min-max scale initialization
   188→        shape = get_scale_shape(orig_weight, orig_layer.group_size)
   189→        self._init_params("min_scale", p_dtype, shape, 1.0, (self.enable_minmax_tuning and self.orig_layer.bits < 16))
   190→        self._init_params("max_scale", p_dtype, shape, 1.0, (self.enable_minmax_tuning and self.orig_layer.bits < 16))
   191→
   192→        self.weight_quant_func, self.data_type = get_quant_func(
   193→            orig_layer.data_type,
   194→            orig_layer.bits,
   195→            orig_layer.sym,
   196→            self.disable_opt_rtn,
   197→            orig_layer.group_size,
   198→            iters=orig_layer.iters,
   199→        )
   200→        if self.enable_torch_compile:
   201→            self.weight_quant_func = compile_func(self.weight_quant_func, self.device)
   202→
   203→        if self.enable_act_quant:
   204→            self.act_quant_func, self.act_data_type = get_quant_func(
   205→                orig_layer.act_data_type,
   206→                orig_layer.act_bits,
   207→                orig_layer.act_sym,
   208→                disable_opt_rtn=True,
   209→                iters=orig_layer.iters,
   210→            )
   211→            if self.enable_torch_compile:
   212→                self.act_quant_func = compile_func(self.act_quant_func, self.device)
   213→            self._init_params(
   214→                "act_max_scale", p_dtype, (1), 1.0, envs.AR_ENABLE_ACT_MINMAX_TUNING or (not orig_layer.act_dynamic)
   215→            )
   216→            self._init_params("act_min_scale", p_dtype, (1), 1.0, envs.AR_ENABLE_ACT_MINMAX_TUNING)
   217→
   218→        # Bias tuning
   219→        if self.enable_norm_bias_tuning:
   220→            self._init_params("bias_v", p_dtype, self.orig_layer.bias.shape, 0, True)
   221→            from auto_round.data_type.int import quant_tensor_asym_wo_round
   222→
   223→            self.bias_quant_func = quant_tensor_asym_wo_round
   224→            self.params["bias_v"] = self.bias_v

   226→    def _init_params(self, name, dtype, shape, value, tunable):
   227→        """Initializes a parameter for tuning or uses a constant if tuning is disabled.
   228→
   229→        Args:
   230→            name (str): Name of the parameter.
   231→            dtype (torch.dtype): Data type of the parameter.
   232→            shape (tuple): Shape of the parameter.
   233→            value (float): Initial value for the parameter.
   234→            tunable (bool): Whether the parameter should be tunable.
   235→        """
   236→        if tunable:
   237→            p = torch.nn.Parameter(torch.ones(shape, device=self.device, dtype=dtype) * value, requires_grad=True)
   238→            self.params.update({name: p})
   239→        else:
   240→            p = torch.tensor(1.0 * value, device=self.device, dtype=dtype)
   241→
   242→        setattr(self, name, p)

   244→    def _qdq_weight(self, value, min_scale, max_scale):
   245→        """Quantizes and dequantizes weights with tuning parameters.
   246→
   247→        Args:
   248→            value (torch.Tensor): Value added for rounding for tuning.
   249→            min_scale (torch.Tensor): Minimum scale for the min value of quantization.
   250→            max_scale (torch.Tensor): Maximum scale for the max value of quantization.
   251→
   252→        Returns:
   253→            tuple: Quantized weight, scale, and zero point.
   254→        """
   255→        if self.orig_layer.bits >= 16:
   256→            return self.orig_layer.weight, None, None
   257→        min_bound, max_bound = self.minmax_scale_bound
   258→        min_scale.data.clamp_(min_bound, max_bound)
   259→        max_scale.data.clamp_(min_bound, max_bound)
   260→        weight = self.orig_layer.weight
   261→        if weight.device.type == "meta":
   262→            weight = self.orig_layer.get_weight().to(self.device)
   263→        if type(self.orig_layer) == transformers.pytorch_utils.Conv1D:
   264→            weight = weight.t()
   265→
   266→        quant_kwargs = {}
   267→        if hasattr(self.orig_layer, "super_bits"):
   268→            quant_kwargs["super_bits"] = self.orig_layer.super_bits
   269→            quant_kwargs["super_group_size"] = self.orig_layer.super_group_size
   270→        if hasattr(self, "_extra_quant_kwargs"):
   271→            quant_kwargs.update(self._extra_quant_kwargs())
   272→
   273→        weight_q, scale, zp = self.weight_quant_func(
   274→            weight.to(self.device),
   275→            bits=self.orig_layer.bits,
   276→            group_size=self.orig_layer.group_size,
   277→            v=value,
   278→            min_scale=min_scale,
   279→            max_scale=max_scale,
   280→            scale_dtype=self.orig_layer.scale_dtype,
   281→            tensor_min=self.weight_min,
   282→            tensor_max=self.weight_max,
   283→            data_type=self.data_type,
   284→            q_scale_thresh=self.q_scale_thresh,
   285→            imatrix=self.orig_layer.imatrix.to(weight.device) if hasattr(self.orig_layer, "imatrix") else None,
   286→            global_scale=getattr(self, "weight_global_scale", None),
   287→            init_scale=getattr(self, "init_scale", None),
   288→            **quant_kwargs,
   289→        )
   290→        weight_q = weight_q.to(weight.dtype)
   291→        if type(self.orig_layer) == transformers.pytorch_utils.Conv1D:
   292→            weight_q = weight_q.t()
   293→        return weight_q, scale, zp
   294→
   295→    def _qdq_act(self, x, act_min_scale=torch.tensor(1.0), act_max_scale=torch.tensor(1.0), act_max=None):
   296→        """Quantizes and dequantizes activations.
   297→
   298→        Args:
   299→            x (torch.Tensor): Input activations.
   300→            act_max_scale (torch.Tensor): Maximum scale for the act_max
   301→            act_max (torch.Tensor, optional): Maximum value for activation quantization. Defaults to None.
   302→
   303→        Returns:
   304→            tuple: Quantized activation, scale, and zero point.
   305→        """
   306→        act_max_scale.data.clamp_(0, 1.0)
   307→        act_min_scale.data.clamp_(0, 1.0)
   308→        env_act_scale = envs.AR_ACT_SCALE  # fixed activation ratio,prioritize to use this one if set
   309→        x, scale, zp = self.act_quant_func(
   310→            x,
   311→            bits=self.orig_layer.act_bits,
   312→            group_size=self.orig_layer.act_group_size,
   313→            scale_dtype=self.orig_layer.scale_dtype,
   314→            q_scale_thresh=self.q_scale_thresh,
   315→            data_type=self.act_data_type,
   316→            tensor_max=act_max,  # for static
   317→            max_scale=act_max_scale if math.isclose(env_act_scale, 1.0, rel_tol=1e-6) else env_act_scale,
   318→            min_scale=act_min_scale if math.isclose(env_act_scale, 1.0, rel_tol=1e-6) else env_act_scale,
   319→            global_scale=getattr(self, "input_global_scale", None),
   320→        )
   321→        return x, scale, zp
   322→
   323→    def _qdq_bias(self, bias, bias_v):
   324→        """Quantizes and dequantizes bias.
   325→
   326→        Args:
   327→            bias (torch.Tensor): Bias tensor to be quantized.
   328→            bias_v (torch.Tensor): Value added for rounding for tuning.
   329→
   330→        Returns:
   331→            tuple: Quantized bias, scale, and zero point.
   332→        """
   333→        bias_bits = 4  ## hard code
   334→        bias_group_size = -1
   335→        bias, scale, zp = self.bias_quant_func(
   336→            bias,
   337→            bits=bias_bits,
   338→            group_size=bias_group_size,
   339→            v=bias_v,
   340→            q_scale_thresh=self.q_scale_thresh,
   341→            global_scale=getattr(self, "weight_global_scale", None),
   342→        )
   343→        return bias, scale, zp

   345→    def unwrapper(self, best_params):
   346→        """Restores the original layer by applying the best tuning parameters.
   347→
   348→        Args:
   349→            best_params (dict): Dictionary containing the best tuning parameters.
   350→
   351→        Returns:
   352→            torch.nn.Module: The unwrapped and restored original layer.
   353→        """
   354→
   355→        def _preserve_global_name(layer):
   356→            if hasattr(self.orig_layer, "global_name") and not hasattr(layer, "global_name"):
   357→                layer.global_name = self.orig_layer.global_name
   358→            return layer
   359→
   360→        best_params = best_params or {}
   361→        v = best_params.get("value", torch.tensor(0.0)).to(self.device)
   362→        min_scale = best_params.get("min_scale", torch.tensor(1.0)).to(self.device)
   363→        max_scale = best_params.get("max_scale", torch.tensor(1.0)).to(self.device)
   364→
   365→        if self.orig_layer.weight.device.type == "meta":
   366→            self.orig_layer.to(self.device)
   367→        # Unwrapper weight
   368→        qdq_weight, scale, zp = self._qdq_weight(v, min_scale, max_scale)
   369→        # if hasattr(self.orig_layer, "imatrix"):
   370→        #     self.orig_layer.imatrix = None
   371→        self.orig_layer.weight.data.copy_(qdq_weight)
   372→        self.orig_layer.weight.grad = None
   373→
   374→        shape = qdq_weight.shape
   375→        if type(self.orig_layer) == transformers.pytorch_utils.Conv1D:
   376→            shape = qdq_weight.t().shape
   377→
   378→        def _set_dict_attr(attr_dict, attr_name):
   379→            for key in attr_dict.keys():
   380→                if key == attr_name:
   381→                    setattr(self.orig_layer, attr_name, attr_dict[key].reshape(shape[0], -1).to("cpu"))
   382→                else:
   383→                    name = "w_" + key
   384→                    setattr(self.orig_layer, name, attr_dict[key].to("cpu"))
   385→
   386→        if not isinstance(self.orig_layer.group_size, tuple):
   387→            if isinstance(scale, dict):
   388→                _set_dict_attr(scale, "scale")
   389→            elif scale is None:
   390→                self.orig_layer.scale = None
   391→            elif scale.numel() > 1:
   392→                self.orig_layer.scale = scale.reshape(shape[0], -1).to("cpu")
   393→            else:
   394→                self.orig_layer.scale = scale.view(-1).to("cpu")
   395→        else:
   396→            self.orig_layer.scale = scale.to("cpu")
   397→
   398→        if zp is not None:
   399→            if isinstance(zp, dict):
   400→                _set_dict_attr(zp, "zp")
   401→            elif isinstance(zp, torch.Tensor):
   402→                if zp.numel() > 1:
   403→                    zp = zp.reshape(shape[0], -1)
   404→                    self.orig_layer.zp = zp.to("cpu")
   405→                else:
   406→                    self.orig_layer.zp = zp.view(-1).to("cpu")
   407→            else:
   408→                self.orig_layer.zp = zp
   409→        else:
   410→            self.orig_layer.zp = None
   411→
   412→        if self.weight_global_scale is not None:
   413→            global_scale = self.weight_global_scale
   414→            assert global_scale.numel() == 1
   415→            self.orig_layer.weight_global_scale = global_scale.to("cpu")
   416→
   417→        # Unwrapper bias
   418→        if self.enable_norm_bias_tuning and "bias_v" in best_params.keys():  ##fake quant
   419→            bias_v = best_params["bias_v"].to(self.device)
   420→            bias = self.orig_layer.bias
   421→            if bias is not None and bias.device.type == "meta":
   422→                bias = self.orig_layer.get_bias().to(self.device)
   423→            bias, _, _ = self._qdq_bias(bias, bias_v)
   424→            self.orig_layer.bias.grad = None
   425→            self.orig_layer.bias.data.copy_(bias)
   426→
   427→        if hasattr(self.orig_layer, "update"):
   428→            self.orig_layer.update()
   429→            self.orig_layer.to("meta")
   430→
   431→        # Unwrapper act
   432→        if self.enable_act_quant:
   433→            if not self.orig_layer.act_dynamic:
   434→                act_max_scale = best_params.get("act_max_scale", torch.tensor(1.0)).to(self.device)
   435→                act_max = self.orig_layer.act_max if hasattr(self.orig_layer, "act_max") else None
   436→                if act_max is not None:
   437→                    tmp_shape = 1
   438→                    if self.orig_layer.act_group_size > 1:
   439→                        tmp_shape = (act_max.shape[0], self.orig_layer.act_group_size)
   440→                    elif self.orig_layer.act_group_size == -1:
   441→                        tmp_shape = (act_max.shape[0], 1)
   442→                    _, act_scale, _ = self._qdq_act(
   443→                        torch.zeros(tmp_shape).to(self.device),
   444→                        act_min_scale=self.act_min_scale,
   445→                        act_max_scale=self.act_max_scale,
   446→                        act_max=act_max,
   447→                    )
   448→                    self.orig_layer.act_max = self.orig_layer.act_max * act_max_scale.item()
   449→                    self.orig_layer.act_max = self.orig_layer.act_max.to("cpu")
   450→                else:
   451→                    act_scale = torch.ones(1, dtype=self.orig_layer.scale_dtype)
   452→                self.orig_layer.act_scale = act_scale.to("cpu")
   453→
   454→            self.orig_layer.q_scale_thresh = self.q_scale_thresh
   455→            self.orig_layer.data_type = self.data_type
   456→            self.orig_layer.act_min_scale = self.act_min_scale
   457→            self.orig_layer.act_max_scale = self.act_max_scale
   458→
   459→            self.orig_layer.act_data_type = self.act_data_type
   460→            self.orig_layer.act_quant_func = self.act_quant_func
   461→            wrapper_layer = WrapperWALayer(
   462→                self.orig_layer,
   463→                enable_torch_compile=self.enable_torch_compile,
   464→                device=self.device,
   465→            )
   466→            return _preserve_global_name(wrapper_layer)
   467→
   468→        return _preserve_global_name(self.orig_layer)

   470→    def linear_forward(self, x, weight, bias):
   471→        """Performs the forward pass for a linear layer.
   472→
   473→        Args:
   474→            x (torch.Tensor): Input tensor.
   475→            weight (torch.Tensor): Weight tensor for the linear layer.
   476→            bias (torch.Tensor): Bias tensor for the linear layer.
   477→
   478→        Returns:
   479→            torch.Tensor: Output tensor after applying the linear layer.
   480→        """
   481→        return F.linear(x, weight, bias)  # pylint: disable=E1102
   482→
   483→    def all_reduce_linear_forward(self, x, weight, bias):
   484→        """Performs the forward pass for a linear layer.
   485→
   486→        Args:
   487→            x (torch.Tensor): Input tensor.
   488→            weight (torch.Tensor): Weight tensor for the linear layer.
   489→            bias (torch.Tensor): Bias tensor for the linear layer.
   490→
   491→        Returns:
   492→            torch.Tensor: Output tensor after applying the linear layer.
   493→        """
   494→        output = torch.matmul(x, weight.transpose(-1, -2))
   495→        if self.mp_group is not None:
   496→            dist.inference_all_reduce(output, group=self.mp_group)
   497→        if bias is not None:
   498→            output += bias
   499→        return output
   500→
   501→    def conv1d_forward(self, x, weight, bias):
   502→        """Performs the forward pass for a Conv1D layer.
   503→
   504→        Args:
   505→            x (torch.Tensor): Input tensor.
   506→            weight (torch.Tensor): Weight tensor for the Conv1D layer.
   507→            bias (torch.Tensor): Bias tensor for the Conv1D layer.
   508→
   509→        Returns:
   510→            torch.Tensor: Output tensor after applying the Conv1D layer.
   511→        """
   512→        size_out = x.size()[:-1] + (self.orig_layer.nf,)
   513→        x = torch.addmm(bias, x.view(-1, x.size(-1)), weight)
   514→        x = x.view(*size_out)
   515→        return x
   516→
   517→    def forward(self, x):
   518→        """Executes the forward pass with quantized weights and optional bias/activation quantization.
   519→
   520→        Args:
   521→            x (torch.Tensor): Input tensor.
   522→
   523→        Returns:
   524→            torch.Tensor: Output tensor after applying the wrapped layer.
   525→        """
   526→        # logger.info(self.orig_layer.global_name)
   527→        x = x.to(self.device)
   528→        weight_q, *_ = self._qdq_weight(self.value, self.min_scale, self.max_scale)
   529→
   530→        if self.enable_act_quant:
   531→            # Run orig_layer's forward_pre_hooks (e.g., online Hadamard transform)
   532→            # BEFORE activation quantization, to match inference behavior in WrapperWALayer.
   533→            for hook in self.orig_layer._forward_pre_hooks.values():
   534→                result = hook(self.orig_layer, (x,))
   535→                if result is not None:
   536→                    x = result[0] if isinstance(result, tuple) else result
   537→            act_max = self.orig_layer.act_max if hasattr(self.orig_layer, "act_max") else None
   538→            x, _, _ = self._qdq_act(
   539→                x, act_max_scale=self.act_max_scale, act_min_scale=self.act_min_scale, act_max=act_max
   540→            )
   541→        elif len(self.orig_layer._forward_pre_hooks) > 0:
   542→            # Even without act_quant, run pre-hooks for online Hadamard correctness
   543→            # (when fuse_online_to_weight=False, pre-hooks transform activations).
   544→            for hook in self.orig_layer._forward_pre_hooks.values():
   545→                result = hook(self.orig_layer, (x,))
   546→                if result is not None:
   547→                    x = result[0] if isinstance(result, tuple) else result
   548→
   549→        # pylint: disable=not-callable
   550→        bias = self.orig_layer.bias
   551→        if bias is not None and bias.device.type == "meta":
   552→            bias = self.orig_layer.get_bias().to(self.device)
   553→        if self.enable_norm_bias_tuning:
   554→            bias, _, _ = self._qdq_bias(bias, self.bias_v)
   555→
   556→        output = self.orig_forward(x, weight_q, bias).to(self.output_device)
   557→
   558→        # Execute post-hooks from orig_layer (e.g., v_proj per-head Hadamard
   559→        # when online rotation is not fused into weights).
   560→        for hook in self.orig_layer._forward_hooks.values():
   561→            hook_result = hook(self.orig_layer, (x,), output)
   562→            if hook_result is not None:
   563→                output = hook_result
   564→
   565→        return output
   566→

   568→class WrapperWALayer(torch.nn.Module):
   569→
   570→    def __init__(self, orig_layer, enable_torch_compile=True, device="cpu"):
   571→        super(WrapperWALayer, self).__init__()
   572→        self.orig_layer = orig_layer
   573→        self.enable_torch_compile = enable_torch_compile
   574→        self.device = device
   575→        self.data_type = orig_layer.data_type if hasattr(orig_layer, "data_type") else None
   576→        self.act_data_type = orig_layer.act_data_type if hasattr(orig_layer, "act_data_type") else None
   577→        self.act_quant_func = self.orig_layer.act_quant_func
   578→        if self.enable_torch_compile:
   579→            self.act_quant_func = compile_func(self.act_quant_func, self.device)
   580→        self.extra_repr_org = orig_layer.extra_repr
   581→
   582→        # Steal forward_pre_hooks from orig_layer (e.g., Hadamard transform hooks)
   583→        # and remove them from orig_layer so they won't fire again inside orig_layer.forward().
   584→        # We will run them explicitly in our forward() BEFORE activation quantization.
   585→        self._stolen_pre_hooks = list(orig_layer._forward_pre_hooks.values())
   586→        orig_layer._forward_pre_hooks.clear()
   587→
   588→    @property
   589→    def weight(self):
   590→        """Exposes the weight of the wrapped layer for external access."""
   591→        return self.orig_layer.weight
   592→
   593→    @property
   594→    def bias(self):
   595→        """Exposes the bias of the wrapped layer for external access."""
   596→        return self.orig_layer.bias
   597→
   598→    def forward(self, x):
   599→        # 1) Run stolen pre_hooks first (e.g., online Hadamard) → smooths activation
   600→        for hook in self._stolen_pre_hooks:
   601→            result = hook(self.orig_layer, (x,))
   602→            if result is not None:
   603→                x = result[0] if isinstance(result, tuple) else result
   604→
   605→        # 2) Activation quantization on the smoothed activation
   606→        import auto_round.envs as envs
   607→
   608→        act_scale = envs.AR_ACT_SCALE
   609→        act_max = self.orig_layer.act_max if hasattr(self.orig_layer, "act_max") else None
   610→
   611→        max_scale = self.orig_layer.act_max_scale if math.isclose(act_scale, 1.0, rel_tol=1e-6) else act_scale
   612→        min_scale = self.orig_layer.act_min_scale if math.isclose(act_scale, 1.0, rel_tol=1e-6) else act_scale
   613→        if act_max is None:
   614→            x, _, _ = self.orig_layer.act_quant_func(
   615→                x,
   616→                bits=self.orig_layer.act_bits,
   617→                group_size=self.orig_layer.act_group_size,
   618→                scale_dtype=self.orig_layer.scale_dtype,
   619→                q_scale_thresh=self.orig_layer.q_scale_thresh,
   620→                data_type=self.orig_layer.act_data_type,
   621→                min_scale=min_scale,
   622→                max_scale=max_scale,
   623→            )
   624→        else:
   625→            x, _, _ = self.orig_layer.act_quant_func(
   626→                x,
   627→                bits=self.orig_layer.act_bits,
   628→                group_size=self.orig_layer.act_group_size,
   629→                scale_dtype=self.orig_layer.scale_dtype,
   630→                q_scale_thresh=self.orig_layer.q_scale_thresh,
   631→                data_type=self.orig_layer.act_data_type,
   632→                act_max=act_max,
   633→            )
   634→        # 3) Linear computation via orig_layer (pre_hooks already removed, no double execution)
   635→        return self.orig_layer.forward(x)
   636→
   637→    def extra_repr(self):
   638→        return f"{self.extra_repr_org()}, weight_type={self.data_type}, act_data_type={self.act_data_type}"

   641→class WrapperLayerNorm(torch.nn.Module):
   642→    """A wrapper for layer normalization with quantized weights.
   643→
   644→    This class wraps a given layer normalization module and applies quantization without round
   645→    to its weights. The quantization is parameterized by the number of bits and
   646→    an optional group size.
   647→    """
   648→
   649→    def __init__(self, orig_layer, bit=4, group_size=-1, device="cpu"):
   650→        super(WrapperLayerNorm, self).__init__()
   651→        self.orig_layer = orig_layer
   652→        self.bits = bit
   653→        self.group_size = group_size
   654→        self.device = self.orig_layer.tuning_device if hasattr(self.orig_layer, "tuning_device") else device
   655→        self.output_device = device
   656→        weight_dtype = torch.float32
   657→        self.q_scale_thresh = 1e-5
   658→        self.v = torch.nn.Parameter(
   659→            reshape_pad_tensor_by_group_size(
   660→                torch.zeros(self.orig_layer.weight.shape, device=self.device, dtype=weight_dtype), self.group_size
   661→            )[0],
   662→            requires_grad=True,
   663→        )
   664→        self.params = {"v": self.v}
   665→        from auto_round.data_type.int import quant_tensor_asym_wo_round
   666→
   667→        self.quant_func = quant_tensor_asym_wo_round
   668→
   669→    def unwrapper(self, best_params):
   670→        if best_params is None:
   671→            return self.orig_layer
   672→        v = best_params["v"].to(self.device)
   673→        weight_q, _, _ = self.quant_func(
   674→            self.orig_layer.weight, self.bits, self.group_size, v, q_scale_thresh=self.q_scale_thresh
   675→        )
   676→        self.orig_layer.q_scale_thresh = self.q_scale_thresh
   677→        self.orig_layer.weight.data.copy_(weight_q)
   678→        return self.orig_layer
   679→
   680→    def forward(self, input):
   681→        input = input.to(self.device)
   682→        weight_q, _, _ = self.quant_func(
   683→            self.orig_layer.weight, self.bits, self.group_size, self.v, q_scale_thresh=self.q_scale_thresh
   684→        )
   685→        import torch.nn.functional as F
   686→
   687→        return F.layer_norm(
   688→            input, self.orig_layer.normalized_shape, weight_q, self.orig_layer.bias, self.orig_layer.eps
   689→        ).to(self.output_device)

   732→        hidden_states = hidden_states.to(self.device)

   753→class WrapperMultiblock(torch.nn.Module):
   754→    """A wrapper for a list of modules to be act as a single block.
   755→
   756→    Args:
   757→    module_list: The list of modules to wrap.
   758→    """
   759→
   760→    def __init__(self, module_list):
   761→        super(WrapperMultiblock, self).__init__()
   762→        self.layers = torch.nn.ModuleList(module_list)
   763→
   764→    def forward(self, x, *args, **kwargs):
   765→        hidden_states = x
   766→        for idx, decoder_layer in enumerate(self.layers):
   767→            layer_outputs = decoder_layer(hidden_states, *args, **kwargs)
   768→            hidden_states = layer_outputs
   769→            if isinstance(hidden_states, tuple) or isinstance(hidden_states, list):
   770→                hidden_states = layer_outputs[0]
   771→        return hidden_states

   774→def wrapper_block(
   775→    block,
   776→    enable_minmax_tuning,
   777→    enable_norm_bias_tuning,
   778→    enable_torch_compile=True,
   779→    device="cpu",
   780→    wrapper_cls=WrapperLinear,
   781→    **kwargs,
   782→):
   783→    """Wraps the layers in the given block with a custom Wrapper module.
   784→
   785→    Args:
   786→        block: The input block containing linear and conv1d layers to be wrapped.
   787→        enable_minmax_tuning: A boolean indicating whether min-max tuning is enabled.
   788→        enable_norm_bias_tuning: A boolean indicating whether normalization and bias tuning is enabled.
   789→        enable_torch_compile: A boolean indicating whether to enable torch compilation.
   790→        device: The device to which the wrapped layers should be moved.
   791→
   792→    Returns:
   793→        list: A list of names of the wrapped layers and unwrapped layers.
   794→    """
   795→    quantized_layers = []
   796→    unquantized_layers = []
   797→    for n, m in block.named_modules():
   798→        if type(m) in SUPPORTED_LAYER_TYPES:
   799→            if not check_to_quantized(m):
   800→                unquantized_layers.append(n)
   801→                continue
   802→            new_m = wrapper_cls(
   803→                m,
   804→                enable_minmax_tuning=enable_minmax_tuning,
   805→                enable_norm_bias_tuning=enable_norm_bias_tuning,
   806→                enable_torch_compile=enable_torch_compile,
   807→                device=device,
   808→                **kwargs,
   809→            )
   810→            set_module(block, n, new_m)
   811→            quantized_layers.append(n)
   812→
   813→        elif enable_norm_bias_tuning:
   814→            if "norm" in m.__class__.__name__.lower():
   815→                if m.__class__.__name__ in NORM_MAPPING.keys():
   816→                    wrapper_layer_class = NORM_MAPPING[m.__class__.__name__]
   817→                    new_m = wrapper_layer_class(m, device=device)
   818→                    set_module(block, n, new_m)
   819→                elif "RMSNorm" in m.__class__.__name__:
   820→                    logger.warning_once(
   821→                        f"use LlamaRMSNorm to wrap {m.__class__.__name__}, please check the correctness yourself"
   822→                    )
   823→                    wrapper_layer_class = NORM_MAPPING["LlamaRMSNorm"]
   824→                    new_m = wrapper_layer_class(m, device=device)
   825→                    set_module(block, n, new_m)
   826→                else:
   827→                    logger.warning_once(f"{m.__class__.__name__} is not supported")
   828→    return quantized_layers, unquantized_layers
