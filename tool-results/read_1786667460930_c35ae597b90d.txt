    12→class QuantLinear(Module):
    13→    """
    14→    Class to quantize weights of given linear layer
    15→
    16→    Parameters:
    17→    ----------
    18→    weight_bit : int, default 4
    19→        Bitwidth for quantized weights.
    20→    bias_bit : int, default None
    21→        Bitwidth for quantized bias.
    22→    full_precision_flag : bool, default False
    23→        If True, use fp32 and skip quantization
    24→    quant_mode : 'symmetric' or 'asymmetric', default 'symmetric'
    25→        The mode for quantization.
    26→    per_channel : bool, default False
    27→        Whether to use channel-wise quantization.
    28→    fix_flag : bool, default False
    29→        Whether the module is in fixed mode or not.
    30→    weight_percentile : float, default 0
    31→        The percentile to setup quantization range, 0 means no use of percentile, 99.9 means to cut off 0.1%.
    32→    """
    33→
    34→    def __init__(self,
    35→                 weight_bit=4,
    36→                 bias_bit=None,
    37→                 full_precision_flag=False,
    38→                 quant_mode='symmetric',
    39→                 per_channel=False,
    40→                 fix_flag=False,
    41→                 weight_percentile=0,
    42→                 ):
    43→        super(QuantLinear, self).__init__()
    44→        self.full_precision_flag = full_precision_flag
    45→        self.weight_bit = weight_bit
    46→        self.quant_mode = quant_mode
    47→        self.per_channel = per_channel
    48→        self.fix_flag = fix_flag
    49→        self.weight_percentile = weight_percentile
    50→        self.bias_bit = bias_bit
    51→        self.quantize_bias = (False if bias_bit is None else True)
    52→        self.quant_mode = quant_mode
    53→        self.counter = 0
    54→
    55→    def __repr__(self):
    56→        s = super(QuantLinear, self).__repr__()
    57→        s = "(" + s + " weight_bit={}, full_precision_flag={}, quantize_fn={})".format(
    58→            self.weight_bit, self.full_precision_flag, self.quant_mode)
    59→        return s
    60→
    61→    def set_param(self, linear):
    62→        self.in_features = linear.in_features
    63→        self.out_features = linear.out_features
    64→        self.register_buffer('fc_scaling_factor', torch.zeros(self.out_features))
    65→        self.weight = Parameter(linear.weight.data.clone())
    66→        self.register_buffer('weight_integer', torch.zeros_like(self.weight))
    67→        self.register_buffer('bias_integer', torch.zeros_like(linear.bias))
    68→        try:
    69→            self.bias = Parameter(linear.bias.data.clone())
    70→        except AttributeError:
    71→            self.bias = None
    72→
    73→    def fix(self):
    74→        self.fix_flag = True
    75→
    76→    def unfix(self):
    77→        self.fix_flag = False
    78→
    79→    def forward(self, x, prev_act_scaling_factor=None):
    80→        """
    81→        using quantized weights to forward activation x
    82→        """
    83→        if type(x) is tuple:
    84→            prev_act_scaling_factor = x[1]
    85→            x = x[0]
    86→
    87→        if self.quant_mode == "symmetric":
    88→            self.weight_function = SymmetricQuantFunction.apply
    89→        elif self.quant_mode == "asymmetric":
    90→            self.weight_function = AsymmetricQuantFunction.apply
    91→        else:
    92→            raise ValueError("unknown quant mode: {}".format(self.quant_mode))
    93→
    94→        w = self.weight
    95→        w_transform = w.data.detach()
    96→        # calculate the quantization range of weights and bias
    97→        if self.per_channel:
    98→            w_min, _ = torch.min(w_transform, dim=1, out=None)
    99→            w_max, _ = torch.max(w_transform, dim=1, out=None)
   100→            if self.quantize_bias:
   101→                b_min = self.bias.data
   102→                b_max = self.bias.data
   103→        else:
   104→            w_min = w_transform.min().expand(1)
   105→            w_max = w_transform.max().expand(1)
   106→            if self.quantize_bias:
   107→                b_min = self.bias.data.min()
   108→                b_max = self.bias.data.max()
   109→
   110→        # perform the quantization
   111→        if not self.full_precision_flag:
   112→            if self.quant_mode == 'symmetric':
   113→                self.fc_scaling_factor = symmetric_linear_quantization_params(self.weight_bit, w_min, w_max,
   114→                                                                              self.per_channel)
   115→                self.weight_integer = self.weight_function(self.weight, self.weight_bit, self.fc_scaling_factor)
   116→
   117→                bias_scaling_factor = self.fc_scaling_factor.view(1, -1) * prev_act_scaling_factor.view(1, -1)
   118→                self.bias_integer = self.weight_function(self.bias, self.bias_bit, bias_scaling_factor)
   119→            else:
   120→                raise Exception('For weight, we only support symmetric quantization.')
   121→        else:
   122→            w = self.weight
   123→            b = self.bias
   124→
   125→        prev_act_scaling_factor = prev_act_scaling_factor.view(1, -1)
   126→        x_int = x / prev_act_scaling_factor
   127→        correct_output_scale = bias_scaling_factor[0].view(1, -1)
   128→
   129→        return ste_round.apply(
   130→            F.linear(x_int, weight=self.weight_integer, bias=self.bias_integer)) * correct_output_scale

   133→class QuantAct(Module):
   134→    """
   135→    Class to quantize given activations
   136→
   137→    Parameters:
   138→    ----------
   139→    activation_bit : int, default 4
   140→        Bitwidth for quantized activations.
   141→    act_range_momentum : float, default 0.95
   142→        Momentum for updating the activation quantization range.
   143→    full_precision_flag : bool, default False
   144→        If True, use fp32 and skip quantization
   145→    running_stat : bool, default True
   146→        Whether to use running statistics for activation quantization range.
   147→    quant_mode : 'symmetric' or 'asymmetric', default 'symmetric'
   148→        The mode for quantization.
   149→    fix_flag : bool, default False
   150→        Whether the module is in fixed mode or not.
   151→    act_percentile : float, default 0
   152→        The percentile to setup quantization range, 0 means no use of percentile, 99.9 means to cut off 0.1%.
   153→    fixed_point_quantization : bool, default False
   154→        Whether to skip deployment-oriented operations and use fixed-point rather than integer-only quantization.
   155→    """
   156→
   157→    def __init__(self,
   158→                 activation_bit=4,
   159→                 act_range_momentum=0.95,
   160→                 full_precision_flag=False,
   161→                 running_stat=True,
   162→                 quant_mode="symmetric",
   163→                 fix_flag=False,
   164→                 act_percentile=0,
   165→                 fixed_point_quantization=False):
   166→        super(QuantAct, self).__init__()
   167→
   168→        self.activation_bit = activation_bit
   169→        self.act_range_momentum = act_range_momentum
   170→        self.full_precision_flag = full_precision_flag
   171→        self.running_stat = running_stat
   172→        self.quant_mode = quant_mode
   173→        self.fix_flag = fix_flag
   174→        self.act_percentile = act_percentile
   175→        self.fixed_point_quantization = fixed_point_quantization
   176→
   177→        self.register_buffer('x_min', torch.zeros(1))
   178→        self.register_buffer('x_max', torch.zeros(1))
   179→        self.register_buffer('act_scaling_factor', torch.zeros(1))
   180→
   181→        self.register_buffer('pre_weight_scaling_factor', torch.ones(1))
   182→        self.register_buffer('identity_weight_scaling_factor', torch.ones(1))
   183→
   184→    def __repr__(self):
   185→        return "{0}(activation_bit={1}, " \
   186→               "full_precision_flag={2}, quant_mode={3}, Act_min: {4:.2f}, " \
   187→               "Act_max: {5:.2f})".format(self.__class__.__name__, self.activation_bit,
   188→                                          self.full_precision_flag, self.quant_mode, self.x_min.item(),
   189→                                          self.x_max.item())
   190→
   191→    def fix(self):
   192→        """
   193→        fix the activation range by setting running stat to False
   194→        """
   195→        self.running_stat = False
   196→        self.fix_flag = True
   197→
   198→    def unfix(self):
   199→        """
   200→        unfix the activation range by setting running stat to True
   201→        """
   202→        self.running_stat = True
   203→        self.fix_flag = False
   204→
   205→    def forward(self, x, pre_act_scaling_factor=None, pre_weight_scaling_factor=None, identity=None,
   206→                identity_scaling_factor=None, identity_weight_scaling_factor=None):
   207→        """
   208→        x: the activation that we need to quantize
   209→        pre_act_scaling_factor: the scaling factor of the previous activation quantization layer
   210→        pre_weight_scaling_factor: the scaling factor of the previous weight quantization layer
   211→        identity: if True, we need to consider the identity branch
   212→        identity_scaling_factor: the scaling factor of the previous activation quantization of identity
   213→        identity_weight_scaling_factor: the scaling factor of the weight quantization layer in the identity branch
   214→
   215→        Note that there are two cases for identity branch:
   216→        (1) identity branch directly connect to the input featuremap
   217→        (2) identity branch contains convolutional layers that operate on the input featuremap
   218→        """
   219→        if type(x) is tuple:
   220→            if len(x) == 3:
   221→                channel_num = x[2]
   222→            pre_act_scaling_factor = x[1]
   223→            x = x[0]
   224→
   225→        if self.quant_mode == "symmetric":
   226→            self.act_function = SymmetricQuantFunction.apply
   227→        elif self.quant_mode == "asymmetric":
   228→            self.act_function = AsymmetricQuantFunction.apply
   229→        else:
   230→            raise ValueError("unknown quant mode: {}".format(self.quant_mode))
   231→
   232→        # calculate the quantization range of the activations
   233→        if self.running_stat:
   234→            if self.act_percentile == 0:
   235→                x_min = x.data.min()
   236→                x_max = x.data.max()
   237→            elif self.quant_mode == 'symmetric':
   238→                x_min, x_max = get_percentile_min_max(x.detach().view(-1), 100 - self.act_percentile,
   239→                                                      self.act_percentile, output_tensor=True)
   240→            # Note that our asymmetric quantization is implemented using scaled unsigned integers without zero_points,
   241→            # that is to say our asymmetric quantization should always be after ReLU, which makes
   242→            # the minimum value to be always 0. As a result, if we use percentile mode for asymmetric quantization,
   243→            # the lower_percentile will be set to 0 in order to make sure the final x_min is 0.
   244→            elif self.quant_mode == 'asymmetric':
   245→                x_min, x_max = get_percentile_min_max(x.detach().view(-1), 0, self.act_percentile, output_tensor=True)
   246→
   247→            # Initialization
   248→            if self.x_min == self.x_max:
   249→                self.x_min += x_min
   250→                self.x_max += x_max
   251→
   252→            # use momentum to update the quantization range
   253→            elif self.act_range_momentum == -1:
   254→                self.x_min = min(self.x_min, x_min)
   255→                self.x_max = max(self.x_max, x_max)
   256→            else:
   257→                self.x_min = self.x_min * self.act_range_momentum + x_min * (1 - self.act_range_momentum)
   258→                self.x_max = self.x_max * self.act_range_momentum + x_max * (1 - self.act_range_momentum)
   259→
   260→        # perform the quantization
   261→        if not self.full_precision_flag:
   262→            if self.quant_mode == 'symmetric':
   263→                self.act_scaling_factor = symmetric_linear_quantization_params(self.activation_bit,
   264→                                                                               self.x_min, self.x_max, False)
   265→            # Note that our asymmetric quantization is implemented using scaled unsigned integers
   266→            # without zero_point shift. As a result, asymmetric quantization should be after ReLU,
   267→            # and the self.act_zero_point should be 0.
   268→            else:
   269→                self.act_scaling_factor, self.act_zero_point = asymmetric_linear_quantization_params(
   270→                    self.activation_bit, self.x_min, self.x_max, True)
   271→            if (pre_act_scaling_factor is None) or (self.fixed_point_quantization == True):
   272→                # this is for the case of input quantization,
   273→                # or the case using fixed-point rather than integer-only quantization
   274→                quant_act_int = self.act_function(x, self.activation_bit, self.act_scaling_factor)
   275→            elif type(pre_act_scaling_factor) is list:
   276→                # this is for the case of multi-branch quantization
   277→                branch_num = len(pre_act_scaling_factor)
   278→                quant_act_int = x
   279→                start_channel_index = 0
   280→                for i in range(branch_num):
   281→                    quant_act_int[:, start_channel_index: start_channel_index + channel_num[i], :, :] \
   282→                        = fixedpoint_fn.apply(x[:, start_channel_index: start_channel_index + channel_num[i], :, :],
   283→                                              self.activation_bit, self.quant_mode, self.act_scaling_factor, 0,
   284→                                              pre_act_scaling_factor[i],
   285→                                              pre_act_scaling_factor[i] / pre_act_scaling_factor[i])
   286→                    start_channel_index += channel_num[i]
   287→            else:
   288→                if identity is None:
   289→                    if pre_weight_scaling_factor is None:
   290→                        pre_weight_scaling_factor = self.pre_weight_scaling_factor
   291→                    quant_act_int = fixedpoint_fn.apply(x, self.activation_bit, self.quant_mode,
   292→                                                        self.act_scaling_factor, 0, pre_act_scaling_factor,
   293→                                                        pre_weight_scaling_factor)
   294→                else:
   295→                    if identity_weight_scaling_factor is None:
   296→                        identity_weight_scaling_factor = self.identity_weight_scaling_factor
   297→                    quant_act_int = fixedpoint_fn.apply(x, self.activation_bit, self.quant_mode,
   298→                                                        self.act_scaling_factor, 1, pre_act_scaling_factor,
   299→                                                        pre_weight_scaling_factor,
   300→                                                        identity, identity_scaling_factor,
   301→                                                        identity_weight_scaling_factor)
   302→            correct_output_scale = self.act_scaling_factor.view(-1)
   303→            return (quant_act_int * correct_output_scale, self.act_scaling_factor)
   304→        else:
   305→            return x

   308→class QuantBnConv2d(Module):
   309→    """
   310→    Class to quantize given convolutional layer weights, with support for both folded BN and separate BN.
   311→
   312→    Parameters:
   313→    ----------
   314→    weight_bit : int, default 4
   315→        Bitwidth for quantized weights.
   316→    bias_bit : int, default None
   317→        Bitwidth for quantized bias.
   318→    full_precision_flag : bool, default False
   319→        If True, use fp32 and skip quantization
   320→    quant_mode : 'symmetric' or 'asymmetric', default 'symmetric'
   321→        The mode for quantization.
   322→    per_channel : bool, default False
   323→        Whether to use channel-wise quantization.
   324→    fix_flag : bool, default False
   325→        Whether the module is in fixed mode or not.
   326→    weight_percentile : float, default 0
   327→        The percentile to setup quantization range, 0 means no use of percentile, 99.9 means to cut off 0.1%.
   328→    fix_BN : bool, default False
   329→        Whether to fix BN statistics during training.
   330→    fix_BN_threshold: int, default None
   331→        When to start training with folded BN.
   332→    """
   333→
   334→    def __init__(self,
   335→                 weight_bit=4,
   336→                 bias_bit=None,
   337→                 full_precision_flag=False,
   338→                 quant_mode="symmetric",
   339→                 per_channel=False,
   340→                 fix_flag=False,
   341→                 weight_percentile=0,
   342→                 fix_BN=False,
   343→                 fix_BN_threshold=None):
   344→        super(QuantBnConv2d, self).__init__()
   345→        self.weight_bit = weight_bit
   346→        self.full_precision_flag = full_precision_flag
   347→        self.per_channel = per_channel
   348→        self.fix_flag = fix_flag
   349→        self.weight_percentile = weight_percentile
   350→        self.bias_bit = bias_bit
   351→        self.quantize_bias = False if bias_bit is None else True
   352→        self.quant_mode = quant_mode
   353→        self.fix_BN = fix_BN
   354→        self.training_BN_mode = fix_BN
   355→        self.fix_BN_threshold = fix_BN_threshold
   356→        self.counter = 1
   357→
   358→    def set_param(self, conv, bn):
   359→        self.out_channels = conv.out_channels
   360→        self.register_buffer('convbn_scaling_factor', torch.zeros(self.out_channels))
   361→        self.register_buffer('weight_integer', torch.zeros_like(conv.weight.data))
   362→        self.register_buffer('bias_integer', torch.zeros_like(bn.bias))
   363→
   364→        self.conv = conv
   365→        self.bn = bn
   366→        self.bn.momentum = 0.99
   367→
   368→    def __repr__(self):
   369→        conv_s = super(QuantBnConv2d, self).__repr__()
   370→        s = "({0}, weight_bit={1}, bias_bit={2}, groups={3}, wt-channel-wise={4}, wt-percentile={5}, quant_mode={6})".format(
   371→            conv_s, self.weight_bit, self.bias_bit, self.conv.groups, self.per_channel, self.weight_percentile,
   372→            self.quant_mode)
   373→        return s
   374→
   375→    def fix(self):
   376→        """
   377→        fix the BN statistics by setting fix_BN to True
   378→        """
   379→        self.fix_flag = True
   380→        self.fix_BN = True
   381→
   382→    def unfix(self):
   383→        """
   384→        change the mode (fixed or not) of BN statistics to its original status
   385→        """
   386→        self.fix_flag = False
   387→        self.fix_BN = self.training_BN_mode
   388→
   389→    def forward(self, x, pre_act_scaling_factor=None):
   390→        """
   391→        x: the input activation
   392→        pre_act_scaling_factor: the scaling factor of the previous activation quantization layer
   393→
   394→        """
   395→        if type(x) is tuple:
   396→            pre_act_scaling_factor = x[1]
   397→            x = x[0]
   398→
   399→        if self.quant_mode == "symmetric":
   400→            self.weight_function = SymmetricQuantFunction.apply
   401→        elif self.quant_mode == "asymmetric":
   402→            self.weight_function = AsymmetricQuantFunction.apply
   403→        else:
   404→            raise ValueError("unknown quant mode: {}".format(self.quant_mode))
   405→
   406→        # determine whether to fold BN or not
   407→        if self.fix_flag == False:
   408→            self.counter += 1
   409→            if (self.fix_BN_threshold == None) or (self.counter < self.fix_BN_threshold):
   410→                self.fix_BN = self.training_BN_mode
   411→            else:
   412→                if self.counter == self.fix_BN_threshold:
   413→                    print("Start Training with Folded BN")
   414→                self.fix_BN = True
   415→
   416→        # run the forward without folding BN
   417→        if self.fix_BN == False:
   418→            w_transform = self.conv.weight.data.contiguous().view(self.conv.out_channels, -1)
   419→            w_min = w_transform.min(dim=1).values
   420→            w_max = w_transform.max(dim=1).values
   421→
   422→            conv_scaling_factor = symmetric_linear_quantization_params(self.weight_bit, w_min, w_max, self.per_channel)
   423→            weight_integer = self.weight_function(self.conv.weight, self.weight_bit, conv_scaling_factor)
   424→            conv_output = F.conv2d(x, weight_integer, self.conv.bias, self.conv.stride, self.conv.padding,
   425→                                   self.conv.dilation, self.conv.groups) * conv_scaling_factor.view(1, -1, 1, 1)
   426→
   427→            batch_mean = torch.mean(conv_output, dim=(0, 2, 3))
   428→            batch_var = torch.var(conv_output, dim=(0, 2, 3))
   429→
   430→            # update mean and variance in running stats
   431→            self.bn.running_mean = self.bn.running_mean.detach() * self.bn.momentum + (
   432→                        1 - self.bn.momentum) * batch_mean
   433→            self.bn.running_var = self.bn.running_var.detach() * self.bn.momentum + (1 - self.bn.momentum) * batch_var
   434→
   435→            output_factor = self.bn.weight.view(1, -1, 1, 1) / torch.sqrt(batch_var + self.bn.eps).view(1, -1, 1, 1)
   436→            output = output_factor * (conv_output - batch_mean.view(1, -1, 1, 1)) + self.bn.bias.view(1, -1, 1, 1)
   437→
   438→            return (output, conv_scaling_factor.view(-1) * output_factor.view(-1))
   439→        # fold BN and fix running statistics
   440→        else:
   441→            running_std = torch.sqrt(self.bn.running_var.detach() + self.bn.eps)
   442→            scale_factor = self.bn.weight / running_std
   443→            scaled_weight = self.conv.weight * scale_factor.reshape([self.conv.out_channels, 1, 1, 1])
   444→
   445→            if self.conv.bias is not None:
   446→                scaled_bias = self.conv.bias
   447→            else:
   448→                scaled_bias = torch.zeros_like(self.bn.running_mean)
   449→            scaled_bias = (scaled_bias - self.bn.running_mean.detach()) * scale_factor + self.bn.bias
   450→
   451→            if not self.full_precision_flag:
   452→                if self.per_channel:
   453→                    w_transform = scaled_weight.data.contiguous().view(self.conv.out_channels, -1)
   454→
   455→                    if self.weight_percentile == 0:
   456→                        w_min = w_transform.min(dim=1).values
   457→                        w_max = w_transform.max(dim=1).values
   458→                    else:
   459→                        lower_percentile = 100 - self.weight_percentile
   460→                        upper_percentile = self.weight_percentile
   461→                        input_length = w_transform.shape[1]
   462→
   463→                        lower_index = math.ceil(input_length * lower_percentile * 0.01)
   464→                        upper_index = math.ceil(input_length * upper_percentile * 0.01)
   465→
   466→                        w_min = torch.kthvalue(w_transform, k=lower_index, dim=1).values
   467→                        w_max = torch.kthvalue(w_transform, k=upper_index, dim=1).values
   468→                else:
   469→                    if self.weight_percentile == 0:
   470→                        w_min = scaled_weight.data.min()
   471→                        w_max = scaled_weight.data.max()
   472→                    else:
   473→                        w_min, w_max = get_percentile_min_max(scaled_weight.view(-1), 100 - self.weight_percentile,
   474→                                                              self.weight_percentile, output_tensor=True)
   475→
   476→                if self.quant_mode == 'symmetric':
   477→                    self.convbn_scaling_factor = symmetric_linear_quantization_params(self.weight_bit,
   478→                                                                                      w_min, w_max, self.per_channel)
   479→                    self.weight_integer = self.weight_function(scaled_weight, self.weight_bit,
   480→                                                               self.convbn_scaling_factor)
   481→                    if self.quantize_bias:
   482→                        bias_scaling_factor = self.convbn_scaling_factor.view(1, -1) * pre_act_scaling_factor.view(1,
   483→                                                                                                                   -1)
   484→                        self.bias_integer = self.weight_function(scaled_bias, self.bias_bit, bias_scaling_factor)
   485→                    self.convbn_scaled_bias = scaled_bias
   486→                else:
   487→                    raise Exception('For weight, we only support symmetric quantization.')
   488→
   489→            pre_act_scaling_factor = pre_act_scaling_factor.view(1, -1, 1, 1)
   490→            x_int = x / pre_act_scaling_factor
   491→            correct_output_scale = bias_scaling_factor.view(1, -1, 1, 1)
   492→
   493→            return (F.conv2d(x_int, self.weight_integer, self.bias_integer, self.conv.stride, self.conv.padding,

   569→    """
   570→
   571→    def __init__(self,
   572→                 kernel_size=7,
   573→                 stride=1,
   574→                 padding=0):

   576→
   577→        self.kernel_size = kernel_size
   578→        self.stride = stride
   579→        self.padding = padding

   584→
   585→    def forward(self, x, x_scaling_factor=None):
   586→        if type(x) is tuple:
   587→            x_scaling_factor = x[1]
   588→            x = x[0]
   589→
   590→        if x_scaling_factor is None:
   591→            return self.final_pool(x)
   592→
   593→        x_scaling_factor = x_scaling_factor.view(-1)
   594→        correct_scaling_factor = x_scaling_factor
   595→
   596→        x_int = x / correct_scaling_factor
   597→        x_int = ste_round.apply(x_int)
   598→        x_int = self.final_pool(x_int)
   599→
   600→        x_int = transfer_float_averaging_to_int_averaging.apply(x_int)
   601→
   602→        return (x_int * correct_scaling_factor, correct_scaling_factor)

   605→class QuantConv2d(Module):
   606→    """
   607→    Class to quantize weights of given convolutional layer
   608→
   609→    Parameters:
   610→    ----------
   611→    weight_bit : int, default 4
   612→        Bitwidth for quantized weights.
   613→    bias_bit : int, default None
   614→        Bitwidth for quantized bias.
   615→    full_precision_flag : bool, default False
   616→        If True, use fp32 and skip quantization
   617→    quant_mode : 'symmetric' or 'asymmetric', default 'symmetric'
   618→        The mode for quantization.
   619→    per_channel : bool, default False
   620→        Whether to use channel-wise quantization.
   621→    fix_flag : bool, default False
   622→        Whether the module is in fixed mode or not.
   623→    weight_percentile : float, default 0
   624→        The percentile to setup quantization range, 0 means no use of percentile, 99.9 means to cut off 0.1%.
   625→    """
   626→
   627→    def __init__(self,
   628→                 weight_bit=4,
   629→                 bias_bit=None,
   630→                 full_precision_flag=False,
   631→                 quant_mode="symmetric",
   632→                 per_channel=False,
   633→                 fix_flag=False,
   634→                 weight_percentile=0):
   635→        super(QuantConv2d, self).__init__()
   636→        self.full_precision_flag = full_precision_flag
   637→        self.weight_bit = weight_bit
   638→        self.quant_mode = quant_mode
   639→        self.per_channel = per_channel
   640→        self.fix_flag = fix_flag
   641→        self.weight_percentile = weight_percentile
   642→        self.bias_bit = bias_bit
   643→        self.quantize_bias = (False if bias_bit is None else True)
   644→
   645→    def __repr__(self):
   646→        s = super(QuantConv2d, self).__repr__()
   647→        s = "(" + s + " weight_bit={}, full_precision_flag={}, quant_mode={})".format(self.weight_bit,
   648→                                                                                      self.full_precision_flag,
   649→                                                                                      self.quant_mode)
   650→        return s
   651→
   652→    def set_param(self, conv):
   653→        self.in_channels = conv.in_channels
   654→        self.out_channels = conv.out_channels
   655→        self.kernel_size = conv.kernel_size
   656→        self.stride = conv.stride
   657→        self.padding = conv.padding
   658→        self.dilation = conv.dilation
   659→        self.groups = conv.groups
   660→        self.conv = conv
   661→        self.register_buffer('conv_scaling_factor', torch.zeros(self.out_channels))
   662→        self.weight = Parameter(conv.weight.data.clone())
   663→        self.register_buffer('weight_integer', torch.zeros_like(self.weight, dtype=torch.int8))
   664→        try:
   665→            self.bias = Parameter(conv.bias.data.clone())
   666→        except AttributeError:
   667→            self.bias = None
   668→
   669→    def fix(self):
   670→        self.fix_flag = True
   671→
   672→    def unfix(self):
   673→        self.fix_flag = False
   674→
   675→    def forward(self, x, pre_act_scaling_factor=None):
   676→        if type(x) is tuple:
   677→            pre_act_scaling_factor = x[1]
   678→            x = x[0]
   679→
   680→        if self.quant_mode == "symmetric":
   681→            self.weight_function = SymmetricQuantFunction.apply
   682→        elif self.quant_mode == "asymmetric":
   683→            self.weight_function = AsymmetricQuantFunction.apply
   684→        else:
   685→            raise ValueError("unknown quant mode: {}".format(self.quant_mode))
   686→
   687→        w = self.weight
   688→
   689→        # calculate quantization range
   690→        if self.per_channel:
   691→            w_transform = w.data.contiguous().view(self.out_channels, -1)
   692→
   693→            if self.weight_percentile == 0:
   694→                w_min = w_transform.min(dim=1).values
   695→                w_max = w_transform.max(dim=1).values
   696→            else:
   697→                lower_percentile = 100 - self.weight_percentile
   698→                upper_percentile = self.weight_percentile
   699→                input_length = w_transform.shape[1]
   700→
   701→                lower_index = math.ceil(input_length * lower_percentile * 0.01)
   702→                upper_index = math.ceil(input_length * upper_percentile * 0.01)
   703→
   704→                w_min = torch.kthvalue(w_transform, k=lower_index, dim=1).values
   705→                w_max = torch.kthvalue(w_transform, k=upper_index, dim=1).values
   706→        elif not self.per_channel:
   707→            if self.weight_percentile == 0:
   708→                w_min = w.data.min()
   709→                w_max = w.data.max()
   710→            else:
   711→                w_min, w_max = get_percentile_min_max(w.view(-1), 100 - self.weight_percentile,
   712→                                                      self.weight_percentile, output_tensor=True)
   713→        # perform quantization
   714→        if self.quant_mode == 'symmetric':
   715→            self.conv_scaling_factor = symmetric_linear_quantization_params(self.weight_bit, w_min, w_max,
   716→                                                                            self.per_channel)
   717→            self.weight_integer = self.weight_function(self.weight, self.weight_bit, self.conv_scaling_factor)
   718→            bias_scaling_factor = self.conv_scaling_factor.view(1, -1) * pre_act_scaling_factor.view(1, -1)
   719→            if self.quantize_bias and (self.bias is not None):
   720→                self.bias_integer = self.weight_function(self.bias, self.bias_bit, bias_scaling_factor)
   721→            else:
   722→                self.bias_integer = None
   723→        else:
   724→            raise Exception('For weight, we only support symmetric quantization.')
   725→
   726→        pre_act_scaling_factor = pre_act_scaling_factor.view(1, -1, 1, 1)
   727→        x_int = x / pre_act_scaling_factor
   728→        correct_output_scale = bias_scaling_factor.view(1, -1, 1, 1)
   729→
   730→        if self.bias is None:
   731→            return (F.conv2d(x_int, self.weight_integer, torch.zeros_like(bias_scaling_factor.view(-1)),
   732→                             self.conv.stride, self.conv.padding, self.conv.dilation, self.conv.groups)
   733→                    * correct_output_scale, self.conv_scaling_factor)
   734→        else:
   735→            return (F.conv2d(x_int, self.weight_integer, self.bias_integer, self.conv.stride, self.conv.padding,
   736→                             self.conv.dilation, self.conv.groups) * correct_output_scale, self.conv_scaling_factor)
   737→

   739→def freeze_model(model):
   740→    """
   741→    freeze the activation range
   742→    """
   743→    if type(model) == QuantAct:
   744→        model.fix()
   745→    elif type(model) == QuantConv2d:
   746→        model.fix()
   747→    elif type(model) == QuantLinear:
   748→        model.fix()
   749→    elif type(model) == QuantBnConv2d:
   750→        model.fix()
   751→    elif type(model) == nn.Sequential:
   752→        for n, m in model.named_children():
   753→            freeze_model(m)
   754→    else:
   755→        for attr in dir(model):
   756→            mod = getattr(model, attr)
   757→            if isinstance(mod, nn.Module) and 'norm' not in attr:
   758→                freeze_model(mod)
   759→
   760→
   761→def unfreeze_model(model):
   762→    """
   763→    unfreeze the activation range
   764→    """
   765→    if type(model) == QuantAct:
   766→        model.unfix()
   767→    elif type(model) == QuantConv2d:
   768→        model.unfix()
   769→    elif type(model) == QuantLinear:
   770→        model.unfix()
   771→    elif type(model) == QuantBnConv2d:
   772→        model.unfix()
   773→    elif type(model) == nn.Sequential:
   774→        for n, m in model.named_children():
   775→            unfreeze_model(m)
   776→    else:
   777→        for attr in dir(model):
   778→            mod = getattr(model, attr)
   779→            if isinstance(mod, nn.Module) and 'norm' not in attr:
   780→                unfreeze_model(mod)
