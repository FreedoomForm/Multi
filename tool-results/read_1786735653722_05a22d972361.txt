    29→    29→class QuantConfig:
    30→    30→    """
    31→    31→    Quantization configuration for Linear layers.
    32→    32→    """
    33→    33→
    34→    34→    def __init__(self, bit_config_map: Dict[int, Dict[str, Union[int, bool,
    35→    35→                                                                 List[int]]]]):
    36→    36→        """
    37→    37→        {
    38→    38→            bit_width: {
    39→    39→                'group_size': int, 
    40→    40→                'asymmetric': bool, 
    41→    41→                'gptq': bool,         # optional
    42→    42→                'clip_shrink': bool,  # optional
    43→    43→                'indices': List[int]  # optional for single bit-width.
    44→    44→            }
    45→    45→        }
    46→    46→        """
    47→    47→        self.bit_config_map = bit_config_map
    48→    48→
    49→    49→    def verify(self, weight_shape: Tuple[int, int]):
    50→    50→        for bit_width, config in self.bit_config_map.items():
    51→    51→            assert bit_width > 0
    52→    52→            assert 'group_size' in config
    53→    53→            assert 'asymmetric' in config
    54→    54→        if len(self.bit_config_map) > 1:
    55→    55→            # If there are multiple bit-widths, at least one of them should have
    56→    56→            # indices.
    57→    57→            at_least_one_indices = False
    58→    58→            for _, config in self.bit_config_map.items():
    59→    59→                if 'indices' in config:
    60→    60→                    at_least_one_indices = True
    61→    61→                    break
    62→    62→            assert at_least_one_indices
    63→    63→
    64→    64→            # Check if all indices are covered.
    65→    65→            all_indices = set()
    66→    66→            for _, config in self.bit_config_map.items():
    67→    67→                if 'indices' in config:
    68→    68→                    all_indices.update(config['indices'])
    69→    69→            assert all_indices == set(range(weight_shape[0]))
    70→    70→
    71→    71→    def get_uniform_bit(self, weight_shape: Tuple[int, int]):
    72→    72→        # TODO: rewrite the related logic.
    73→    73→        if len(self.bit_config_map) == 1:
    74→    74→            return next(iter(self.bit_config_map.keys()))
    75→    75→        for bit_width, config in self.bit_config_map.items():
    76→    76→            if 'indices' not in config:
    77→    77→                continue
    78→    78→            if len(config['indices']) == weight_shape[0]:
    79→    79→                return bit_width
    80→    80→        return -1

   150→   150→        super(QuantizedLinearLayer, self).__init__(**kwargs)
   151→   151→        # TODO: add weight quantization here.

   169→   169→        else:
   170→   170→            return nn.functional.linear(input, self.weight)

   173→   173→class Quantizer(torch.autograd.Function):
   174→   174→    __DEBUG = True
   175→   175→
   176→   176→    @classmethod
   177→   177→    def __get_quantize_param_raw(cls, input: torch.Tensor, bit_width: int,
   178→   178→                                 group_size: int, asymmetric: bool):
   179→   179→        '''
   180→   180→        Quantize along the row direction, which is the out_channels direction
   181→   181→        for the weight tensor and the token direction for the activation tensor
   182→   182→        by default.
   183→   183→        '''
   184→   184→
   185→   185→        shape = input.shape
   186→   186→        device = input.device
   187→   187→        if input.dim() == 3:  # activation
   188→   188→            input = input.reshape(-1, shape[-1])
   189→   189→        n_rows, hidden_size = input.shape
   190→   190→        if group_size > 0:
   191→   191→            assert hidden_size % group_size == 0
   192→   192→        else:
   193→   193→            group_size = hidden_size
   194→   194→        input = input.reshape(-1, group_size)
   195→   195→
   196→   196→        if asymmetric:
   197→   197→            min_val, max_val = input.aminmax(dim=1, keepdim=True)
   198→   198→            min_val = min_val.reshape(n_rows, -1)  # not neessary?
   199→   199→            max_val = max_val.reshape(n_rows, -1)
   200→   200→            min_int = 0
   201→   201→            max_int = 2**bit_width - 1
   202→   202→        else:
   203→   203→            min_val = 0
   204→   204→            max_val = input.abs().amax(dim=1, keepdim=True).clamp(min=1e-5)
   205→   205→            max_val = max_val.reshape(n_rows, -1)
   206→   206→            min_int = -(2**(bit_width - 1))
   207→   207→            max_int = 2**(bit_width - 1) - 1
   208→   208→
   209→   209→        return min_val, max_val, min_int, max_int

   211→   211→    @classmethod
   212→   212→    @torch.no_grad()
   213→   213→    def __shrinked_scales_zeros(cls, min_val, max_val, min_int, max_int, ratio,
   214→   214→                                asymmetric):
   215→   215→        min_val, max_val = min_val * ratio, max_val * ratio
   216→   216→        if asymmetric:
   217→   217→            # TODO: test removing the clamp.
   218→   218→            scales = (max_val - min_val).clamp(min=1e-5) / max_int
   219→   219→            zeros = (-torch.round(min_val / scales)).clamp_(min_int, max_int)
   220→   220→        else:
   221→   221→            scales = max_val / max_int
   222→   222→            zeros = 0
   223→   223→        return scales, zeros

   225→   225→    @classmethod
   226→   226→    @torch.no_grad()
   227→   227→    def __quantize_with_param(cls,
   228→   228→                              input: torch.Tensor,
   229→   229→                              bit_width: int,
   230→   230→                              group_size: int,
   231→   231→                              asymmetric: bool,
   232→   232→                              scales: torch.Tensor,
   233→   233→                              zeros: Union[torch.Tensor, int],
   234→   234→                              min_int: int,
   235→   235→                              max_int: int,
   236→   236→                              fake: bool = True):
   237→   237→        shape = input.shape
   238→   238→        input = input.reshape(-1, shape[-1])
   239→   239→        out_features, in_features = input.shape
   240→   240→        if group_size > 0:
   241→   241→            assert in_features % group_size == 0
   242→   242→        else:
   243→   243→            group_size = in_features
   244→   244→        num_groups_per_channel = in_features // group_size
   245→   245→        group_number = out_features * num_groups_per_channel
   246→   246→        input = input.reshape(group_number, group_size)
   247→   247→        scales = scales.reshape(group_number, 1)
   248→   248→        zeros = zeros.reshape(group_number, 1) if isinstance(
   249→   249→            zeros, torch.Tensor) else zeros
   250→   250→        quantized = torch.clamp(
   251→   251→            torch.round(input / scales) + zeros, min_int, max_int)
   252→   252→        if cls.__DEBUG:
   253→   253→            assert torch.isnan(quantized).sum() == 0
   254→   254→            assert torch.isinf(quantized).sum() == 0
   255→   255→
   256→   256→        if fake:
   257→   257→            dequantized = (quantized - zeros) * scales
   258→   258→            return dequantized.reshape(shape)
   259→   259→        else:
   260→   260→            if bit_width == 8:
   261→   261→                dtype = torch.uint8 if asymmetric else torch.int8
   262→   262→            elif bit_width == 4:
   263→   263→                # To support the 4-bit, maybe by packing.
   264→   264→                dtype = torch.uint4 if asymmetric else torch.int4
   265→   265→            else:
   266→   266→                raise ValueError("Unsupported bit-width.")
   267→   267→            quantized = quantized.to(dtype).reshape(shape)
   268→   268→            shape[-1] = num_groups_per_channel
   269→   269→            scales = scales.reshape(shape)
   270→   270→            zeros = zeros.to(dtype).reshape(shape) if asymmetric else zeros
   271→   271→            return quantized, scales, zeros

   273→   273→    @classmethod
   274→   274→    @torch.compile
   275→   275→    @torch.no_grad()
   276→   276→    def __mse_with_delta_weight(cls, input, delta_weight, dim=None):
   277→   277→        mse = ((torch.einsum('ti,oi->to', input,
   278→   278→                             delta_weight)**2).mean(dim=dim))
   279→   279→        return mse

   281→   281→    @classmethod
   282→   282→    @torch.no_grad()
   283→   283→    def __search_clip_param_groupwise(cls, weight, activation, bit_width,
   284→   284→                                      group_size, asymmetric):
   285→   285→        out_features, in_features = weight.shape
   286→   286→        dtype = weight.dtype
   287→   287→        device = weight.device
   288→   288→        if group_size > 0:
   289→   289→            assert in_features % group_size == 0
   290→   290→        else:
   291→   291→            group_size = in_features
   292→   292→
   293→   293→        min_val, max_val, min_int, max_int = cls.__get_quantize_param_raw(
   294→   294→            weight, bit_width, group_size, asymmetric)
   295→   295→        num_groups_per_channel = in_features // group_size
   296→   296→        group_number = out_features * num_groups_per_channel
   297→   297→        min_val = min_val.reshape(group_number, 1) if isinstance(
   298→   298→            min_val, torch.Tensor) else min_val
   299→   299→        max_val = max_val.reshape(group_number, 1)
   300→   300→
   301→   301→        group_min_mse = torch.full((num_groups_per_channel, out_features),
   302→   302→                                   float('inf'),
   303→   303→                                   dtype=dtype,
   304→   304→                                   device=device)
   305→   305→        group_best_pct = torch.full((num_groups_per_channel, out_features),
   306→   306→                                    100,
   307→   307→                                    dtype=torch.int,
   308→   308→                                    device=device)
   309→   309→        # input = activation.reshape(-1, activation.shape[-1])
   310→   310→        # [s, n_g, g] -> [n_g, s, g]
   311→   311→        input_grouped = activation.reshape(-1, num_groups_per_channel,
   312→   312→                                           group_size).permute(1, 0,
   313→   313→                                                               2).contiguous()
   314→   314→        # [o, n_g, g] -> [n_g, o, g]
   315→   315→        weight_grouped = weight.reshape(out_features, num_groups_per_channel,
   316→   316→                                        group_size).permute(1, 0,
   317→   317→                                                            2).contiguous()
   318→   318→
   319→   319→        # use_compile = True
   320→   320→        step = 2
   321→   321→        pct_grid = range(80, 101, step)
   322→   322→        for pct in pct_grid:
   323→   323→            ratio = pct / 100.0
   324→   324→            scales, zeros = cls.__shrinked_scales_zeros(min_val, max_val,
   325→   325→                                                        min_int, max_int, ratio,
   326→   326→                                                        asymmetric)
   327→   327→            quantized = cls.__quantize_with_param(weight,
   328→   328→                                                  bit_width,
   329→   329→                                                  group_size,
   330→   330→                                                  asymmetric,
   331→   331→                                                  scales,
   332→   332→                                                  zeros,
   333→   333→                                                  min_int,
   334→   334→                                                  max_int,
   335→   335→                                                  fake=True)
   336→   336→            if cls.__DEBUG:
   337→   337→                assert quantized.isnan().sum() == 0
   338→   338→                assert quantized.isinf().sum() == 0
   339→   339→
   340→   340→            quantized = weight_grouped - quantized.reshape(
   341→   341→                out_features, num_groups_per_channel, group_size).permute(
   342→   342→                    1, 0, 2).contiguous()
   343→   343→            # # [out_features, num_groups_per_channel]
   344→   344→            # mse = (torch.einsum('tng,ong->ton', input_grouped,
   345→   345→            #                     quantized)**2).mean(dim=0)
   346→   346→            for gid, (input_g,
   347→   347→                      delta_g) in enumerate(zip(input_grouped, quantized)):
   348→   348→                if False:
   349→   349→                    debug.dump_mem("search_clip_param_groupwise before mse")
   350→   350→                    print(f"input_g GB: {debug.tensor_size_gigabytes(input_g)}")
   351→   351→                    print(f"delta_g GB: {debug.tensor_size_gigabytes(delta_g)}")
   352→   352→                # TODO: OOM for llama 70B model.
   353→   353→                # mse = ((torch.einsum('tg,og->to', input_g,
   354→   354→                #                      delta_g)**2).mean(dim=0))
   355→   355→                mse = cls.__mse_with_delta_weight(input_g, delta_g, dim=0)
   356→   356→                if False:
   357→   357→                    debug.dump_mem("search_clip_param_groupwise after mse")
   358→   358→                    print(f"mse GB: {debug.tensor_size_gigabytes(mse)}")
   359→   359→                update_mask = mse < group_min_mse[gid]
   360→   360→                group_min_mse[gid][update_mask] = mse[update_mask]
   361→   361→                group_best_pct[gid][update_mask] = pct
   362→   362→            del quantized
   363→   363→            if cls.__DEBUG:
   364→   364→                assert group_min_mse.isnan().sum() == 0
   365→   365→        del group_min_mse
   366→   366→        del input_grouped
   367→   367→        del weight_grouped
   368→   368→
   369→   369→        # quantize according to the best pct.
   370→   370→        ratio = (group_best_pct / 100.0).to(dtype).t().contiguous().reshape(
   371→   371→            group_number, -1)
   372→   372→        scales, zeros = cls.__shrinked_scales_zeros(min_val, max_val, min_int,
   373→   373→                                                    max_int, ratio, asymmetric)
   374→   374→        scales = scales.reshape(out_features, num_groups_per_channel)
   375→   375→        zeros = zeros.reshape(out_features,
   376→   376→                              num_groups_per_channel) if isinstance(
   377→   377→                                  zeros, torch.Tensor) else zeros
   378→   378→
   379→   379→        enable_fallback = True
   380→   380→        if enable_fallback:
   381→   381→            scales_base, zeros_base, _, _ = cls.__get_quant_param(
   382→   382→                weight, bit_width, group_size, asymmetric)
   383→   383→            quant_base = cls.__quantize_with_param(weight,
   384→   384→                                                   bit_width,
   385→   385→                                                   group_size,
   386→   386→                                                   asymmetric,
   387→   387→                                                   scales_base,
   388→   388→                                                   zeros_base,
   389→   389→                                                   min_int,
   390→   390→                                                   max_int,
   391→   391→                                                   fake=True)
   392→   392→            assert quant_base.isnan().sum() == 0
   393→   393→            assert quant_base.isinf().sum() == 0
   394→   394→            # mse_base = (torch.einsum('ti,oi->to',
   395→   395→            #                          activation.reshape(-1, in_features),
   396→   396→            #                          quant_base - weight)**2).mean()
   397→   397→            mse_base = cls.__mse_with_delta_weight(
   398→   398→                activation.reshape(-1, in_features), quant_base - weight)
   399→   399→            del quant_base
   400→   400→            quant_new = cls.__quantize_with_param(weight,
   401→   401→                                                  bit_width,
   402→   402→                                                  group_size,
   403→   403→                                                  asymmetric,
   404→   404→                                                  scales,
   405→   405→                                                  zeros,
   406→   406→                                                  min_int,
   407→   407→                                                  max_int,
   408→   408→                                                  fake=True)
   409→   409→            assert quant_new.isnan().sum() == 0
   410→   410→            assert quant_new.isinf().sum() == 0
   411→   411→            # mse_new = (torch.einsum('ti,oi->to',
   412→   412→            #                         activation.reshape(-1, in_features),
   413→   413→            #                         quant_new - weight)**2).mean()
   414→   414→            mse_new = cls.__mse_with_delta_weight(
   415→   415→                activation.reshape(-1, in_features), quant_new - weight)
   416→   416→            del quant_new
   417→   417→            # assert mse_new <= mse_base, f"{mse_new} not less than {mse_base}"
   418→   418→
   419→   419→            if mse_new > mse_base:
   420→   420→                # Note the per-group smaller mse does not guarantee the global
   421→   421→                # smaller mse, as the per-group mse is calculated based on the
   422→   422→                # partial value on K dimension.
   423→   423→                scales = scales_base
   424→   424→                zeros = zeros_base
   425→   425→                logger.info(f"clip_shrink fallback for a tensor.")
   426→   426→
   427→   427→        return scales, zeros, min_int, max_int

   477→   477→    @classmethod
   478→   478→    @torch.compile
   479→   479→    def __hessinv(cls, H: torch.Tensor, percdamp: float):
   480→   480→        damp = percdamp * torch.mean(torch.diag(H))
   481→   481→        diag = torch.arange(H.size(0), device=H.device)
   482→   482→        H[diag, diag] += damp
   483→   483→        H = torch.linalg.cholesky(H)
   484→   484→        H = torch.cholesky_inverse(H)
   485→   485→        H = torch.linalg.cholesky(H, upper=True)
   486→   486→        Hinv = H.to(torch.float32)

   492→   492→            cls,
   493→   493→            weight: torch.Tensor,
   494→   494→            activation: torch.Tensor,
   495→   495→            bit_width: int,
   496→   496→            group_size: int,
   497→   497→            asymmetric: bool,
   498→   498→            group_reorder: bool,
   499→   499→            #    reorder: bool,
   500→   500→            #    jit_group_param: bool,
   501→   501→            enable_clip_shrink: bool,
   502→   502→            percdamp: float = 0.01,
   503→   503→            fake: bool = True):

   505→   505→        Partial of the code is based on https://github.com/IST-DASLab/gptq.
   506→   506→        '''
   507→   507→
   508→   508→        device = weight.device
   509→   509→        dtype = weight.dtype
   510→   510→        out_features, in_features = weight.shape
   511→   511→        # Transpose the weight to speedup the GPTQ execution (higher memory
   512→   512→        # efficiency).
   513→   513→        weight_update_t = weight.clone().detach().t().contiguous().to(
   514→   514→            torch.float32)
   515→   515→        if group_size > 0:
   516→   516→            assert in_features % group_size == 0
   517→   517→        else:
   518→   518→            group_size = in_features
   519→   519→        block_size = group_size
   520→   520→
   521→   521→        hess_dtype = torch.float64
   522→   522→        assert activation.dim() == 3, "Only support 3D activation tensor."
   523→   523→        assert activation.size(-1) == in_features
   524→   524→        batch = activation.size(0)
   525→   525→        H = torch.zeros(in_features,
   526→   526→                        in_features,
   527→   527→                        dtype=hess_dtype,
   528→   528→                        device=device)
   529→   529→        for sample in activation:
   530→   530→            # Run different samples separately to save memory.
   531→   531→            tmp = math.sqrt(2 / batch) * sample.to(hess_dtype)
   532→   532→            H += torch.einsum('si,sj->ij', tmp, tmp)
   533→   533→            del tmp
   534→   534→        dead = torch.diag(H) == 0
   535→   535→        H[dead, dead] = 1
   536→   536→        weight_update_t[dead, :] = 0
   537→   537→        del dead
   538→   538→
   539→   539→        num_groups_per_channel = in_features // group_size
   540→   540→
   541→   541→        # if reorder:
   542→   542→        #     perm = torch.argsort(torch.diag(H), descending=True)
   543→   543→        #     weight_update_t = weight_update_t[perm, :]
   544→   544→        #     H = H[perm][:, perm]
   545→   545→        #     perm_inv = torch.argsort(perm)
   546→   546→
   547→   547→        if group_reorder:
   548→   548→            grouped_diag_sum = torch.diag(H).reshape(-1, group_size).sum(1)
   549→   549→            group_perm = torch.argsort(grouped_diag_sum, descending=True)
   550→   550→            if cls.__DEBUG:
   551→   551→                assert (group_perm.dim() == 1) and (group_perm.numel()
   552→   552→                                                    == num_groups_per_channel)
   553→   553→            perm_base = (group_perm * group_size).expand(
   554→   554→                group_size, num_groups_per_channel).t().contiguous()
   555→   555→            offsets = torch.tensor(
   556→   556→                list(range(group_size)),
   557→   557→                device=weight.device).repeat(num_groups_per_channel).reshape(
   558→   558→                    num_groups_per_channel, group_size)
   559→   559→            perm = (perm_base + offsets).reshape(-1)
   560→   560→            del perm_base, offsets
   561→   561→
   562→   562→            weight_update_t = weight_update_t[perm, :]
   563→   563→            H = H[perm][:, perm]
   564→   564→
   565→   565→            perm_inv = torch.argsort(perm)
   566→   566→            group_perm_inv = torch.argsort(group_perm)
   567→   567→
   568→   568→        use_compile = True
   569→   569→        if use_compile:
   570→   570→            Hinv = cls.__hessinv(H, percdamp)
   571→   571→        else:
   572→   572→            damp = torch.tensor(percdamp, dtype=hess_dtype,
   573→   573→                                device=device) * torch.mean(torch.diag(H))
   574→   574→            diag = torch.arange(in_features, device=device)
   575→   575→            H[diag, diag] += damp
   576→   576→            del damp, diag
   577→   577→            H = torch.linalg.cholesky(H)
   578→   578→            H = torch.cholesky_inverse(H)
   579→   579→            H = torch.linalg.cholesky(H, upper=True)
   580→   580→            Hinv = H.to(weight_update_t.dtype)
   581→   581→        del H
   582→   582→
   583→   583→        if asymmetric:
   584→   584→            min_int = 0
   585→   585→            max_int = 2**bit_width - 1
   586→   586→        else:
   587→   587→            min_int = -(2**(bit_width - 1))
   588→   588→            max_int = 2**(bit_width - 1) - 1
   589→   589→
   590→   590→        def quant_slice(input: torch.Tensor, s, z):
   591→   591→            assert (input.dim() == 1) and (s.dim() == 1)
   592→   592→            quantized = torch.clamp(
   593→   593→                torch.round(input / s) + z, min_int, max_int)
   594→   594→            dequantized = (quantized - z) * s
   595→   595→            return dequantized, quantized
   596→   596→
   597→   597→        # Use transposed layout to speedup the memory access.
   598→   598→        quantized = torch.full((in_features, out_features),

   600→   600→                               dtype=dtype,
   601→   601→                               device=device)

   604→   604→        #                                                  activation,
   605→   605→        #                                                  bit_width,
   606→   606→        #                                                  group_size,
   607→   607→        #                                                  asymmetric,
   608→   608→        #                                                  enable_clip_shrink,
   609→   609→        #                                                  fake=False)
   610→   610→        #     scales = scales.reshape(out_features,
   611→   611→        #                             num_groups_per_channel).t().contiguous()
   612→   612→        #     if asymmetric:
   613→   613→        #         zeros = zeros.reshape(out_features,
   614→   614→        #                               num_groups_per_channel).t().contiguous()

   617→   617→        #         param_ready = set()

   619→   619→                            float('nan'),
   620→   620→                            dtype=dtype,
   621→   621→                            device=device)
   622→   622→        if asymmetric:
   623→   623→            zeros = torch.full((num_groups_per_channel, out_features),
   624→   624→                               float('nan'),
   625→   625→                               dtype=dtype,
   626→   626→                               device=device)
   627→   627→        input = activation.reshape(-1, in_features)
   628→   628→        for i1 in range(0, in_features, block_size):
   629→   629→            i2 = min(i1 + block_size, in_features)
   630→   630→            count = i2 - i1
   631→   631→
   632→   632→            weight_block_t = weight_update_t[i1:i2, :].clone().contiguous()
   633→   633→            err_block_t = torch.zeros_like(weight_block_t)
   634→   634→            Hinv_block = Hinv[i1:i2, i1:i2]
   635→   635→
   636→   636→            # We can do this here because block_size == group_size.
   637→   637→
   638→   638→            # if not reorder:
   639→   639→            if enable_clip_shrink:
   640→   640→                input_block = input[:, i1:i2].clone().to(
   641→   641→                    weight_block_t.dtype).contiguous()
   642→   642→                s, z, _, _ = cls.__search_clip_param_groupwise(
   643→   643→                    weight_block_t.t().contiguous(), input_block, bit_width,
   644→   644→                    group_size, asymmetric)
   645→   645→            else:
   646→   646→                s, z, _, _ = cls.__get_quant_param(
   647→   647→                    weight_block_t.t().contiguous(), bit_width, group_size,
   648→   648→                    asymmetric)
   649→   649→            s = s.reshape(-1)
   650→   650→            z = z.reshape(-1) if not isinstance(z, int) else z
   651→   651→
   652→   652→            for i in range(count):
   653→   653→                feature_idx = i1 + i
   654→   654→                # if reorder:
   655→   655→                #     feature_idx_real = perm[feature_idx]
   656→   656→                #     if jit_group_param:
   657→   657→                #         group_idx_real = feature_idx_real // group_size
   658→   658→                #         if group_idx_real not in param_ready:
   659→   659→                #             group_indices = list(
   660→   660→                #                 range(group_idx_real * group_size,
   661→   661→                #                       min((group_idx_real + 1) * group_size)))
   662→   662→                #             group_indices = perm_inv[torch.tensor(group_indices,
   663→   663→                #                                          device=weight.device)]
   664→   664→                #             weight_group_tensor = ...
   665→   665→                #             input_group_tensor = ...
   666→   666→                #     else:
   667→   667→                #         s = scales[feature_idx_real, :]
   668→   668→                #         if asymmetric:
   669→   669→                #             z = zeros[feature_idx_real, :]
   670→   670→                #         else:
   671→   671→                #             z = zeros
   672→   672→
   673→   673→                w = weight_block_t[i, :]
   674→   674→                d = Hinv_block[i, i]
   675→   675→                w_q, q = quant_slice(w, s, z)
   676→   676→                # TODO: to wrap more functions into the compile scope.
   677→   677→                err_feature = (w - w_q) / d
   678→   678→                quantized[feature_idx] = q
   679→   679→                weight_block_t[i:, :] -= torch.einsum('i,j->ji', err_feature,
   680→   680→                                                      Hinv_block[i, i:])
   681→   681→                err_block_t[i, :] = err_feature

   683→   683→                                                    Hinv[i1:i2, i2:])

   685→   685→            group_idx = i1 // block_size

   687→   687→            if asymmetric:
   688→   688→                zeros[group_idx] = z.to(dtype).reshape(out_features)
   689→   689→
   690→   690→        if group_reorder:
   691→   691→            quantized = quantized[perm_inv, :]
   692→   692→            scales = scales[group_perm_inv, :]
   693→   693→            zeros = zeros[group_perm_inv, :] if asymmetric else 0
   694→   694→
   695→   695→        quantized = quantized.t().contiguous()
   696→   696→        scales = scales.t().contiguous()
   697→   697→        zeros = zeros.t().contiguous() if asymmetric else 0
   698→   698→        del Hinv
   699→   699→        del weight_update_t
   700→   700→
   701→   701→        if fake:
   702→   702→            dequantized = cls.__dequantize_weight_tensor(
   703→   703→                quantized, scales, zeros, group_size)
   704→   704→            return dequantized
   705→   705→        else:
   706→   706→            if bit_width == 8:

   708→   708→            elif bit_width == 4:

   710→   710→                dtype = torch.uint8 if asymmetric else torch.int8
   711→   711→            else:
   712→   712→                raise ValueError("Unsupported bit-width.")
   713→   713→            quantized = quantized.to(dtype).reshape(out_features, in_features)
   714→   714→            scales = scales.reshape(out_features, -1).t().contiguous()
   715→   715→            zeros = zeros.to(dtype).reshape(
   716→   716→                out_features, -1).t().contiguous() if asymmetric else zeros
